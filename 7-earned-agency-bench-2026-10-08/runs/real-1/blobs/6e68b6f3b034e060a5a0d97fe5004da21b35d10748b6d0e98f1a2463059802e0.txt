def to_snake(name: str) -> str:
    if not name:
        return ''
    result = []
    for i, ch in enumerate(name):
        if ch.isupper():
            prev = name[i - 1] if i > 0 else None
            nxt = name[i + 1] if i + 1 < len(name) else None
            # Rule 1: underscore between lowercase/digit and capital right after it
            if prev is not None and (prev.islower() or prev.isdigit()):
                result.append('_')
            # Rule 2: underscore before a capital that has another capital before it
            # and a lowercase letter after it
            elif prev is not None and prev.isupper() and nxt is not None and nxt.islower():
                result.append('_')
            result.append(ch.lower())
        else:
            result.append(ch)
    return ''.join(result)
