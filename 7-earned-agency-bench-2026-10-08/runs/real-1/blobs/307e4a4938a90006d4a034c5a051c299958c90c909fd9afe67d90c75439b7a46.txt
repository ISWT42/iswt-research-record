import re


def parse_cents(s: str) -> int:
    if not isinstance(s, str):
        raise ValueError("not a string")
    t = s.strip()
    if t == "":
        raise ValueError("empty")

    sign = 1
    if t.startswith("-"):
        sign = -1
        t = t[1:]

    if t.startswith("$"):
        t = t[1:]

    # Split off fractional part
    if "." in t:
        if t.count(".") != 1:
            raise ValueError("multiple dots")
        int_part, frac_part = t.split(".")
        if not (1 <= len(frac_part) <= 2) or not frac_part.isdigit():
            raise ValueError("bad fraction")
        frac_cents = int(frac_part) * (10 if len(frac_part) == 1 else 1)
    else:
        int_part = t
        frac_cents = 0

    if int_part == "":
        raise ValueError("no integer part")

    # Validate integer part with optional thousands commas
    if "," in int_part:
        # must be groups of exactly three separated by commas
        if not re.fullmatch(r"\d{1,3}(,\d{3})+", int_part):
            raise ValueError("bad grouping")
        digits = int_part.replace(",", "")
    else:
        if not int_part.isdigit():
            raise ValueError("bad integer part")
        digits = int_part

    if digits == "":
        raise ValueError("no digits")

    dollars = int(digits)
    return sign * (dollars * 100 + frac_cents)
