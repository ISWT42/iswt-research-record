import re

def parse_cents(s: str) -> int:
    s = s.strip()
    if not s:
        raise ValueError("Empty string")
    
    # Regex breakdown:
    # ^(-)?           : Optional leading minus
    # \$?             : Optional dollar sign
    # (               : Start of number part
    #   \d{1,3}       : 1-3 digits
    #   (,\d{3})*     : Zero or more groups of comma + 3 digits
    #   |             : OR
    #   \d+           : Just digits (no commas)
    # )
    # (\.\d{1,2})?    : Optional dot followed by 1 or 2 digits
    # $               : End of string
    
    pattern = r'^(-)?\$?(\d{1,3}(,\d{3})*|\d+)(\.\d{1,2})?$'
    match = re.match(pattern, s)
    
    if not match:
        raise ValueError("Invalid money format")
    
    sign_group, main_part, comma_group, decimal_part = match.groups()
    
    # Remove commas from the main part
    clean_main = main_part.replace(',', '')
    
    # Handle decimals
    cents = int(clean_main) * 100
    if decimal_part:
        # Remove the dot
        dec_val = decimal_part[1:]
        # '0.7' -> 70 cents, '0.07' -> 7 cents
        if len(dec_val) == 1:
            cents += int(dec_val) * 10
        else:
            cents += int(dec_val)
            
    if sign_group == '-':
        cents = -cents
        
    return cents
