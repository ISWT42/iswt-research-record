def anagram_groups(words: list) -> list:
    """
    Group words that are anagrams of each other.
    
    Args:
        words: A list of strings to group into anagram groups.
        
    Returns:
        A list of lists, where each inner list contains words that are anagrams
        of each other. Groups are ordered by the first appearance of any word
        in that group in the input list. Words within a group maintain their
        original relative order from the input.
    """
    if not words:
        return []
    
    # Dictionary to map sorted character tuple to list of indices
    # We store indices to preserve original order and handle duplicates correctly
    anagram_map = {}
    
    for i, word in enumerate(words):
        # Normalize word: lowercase and sort characters to create key
        normalized_key = tuple(sorted(word.lower()))
        
        if normalized_key not in anagram_map:
            anagram_map[normalized_key] = []
        
        anagram_map[normalized_key].append(i)
    
    # Now we need to order the groups by the first appearance of any word in that group
    # For each group (key), the "first appearance" is the minimum index among all words in that group
    
    # Create a list of (first_index, group_indices)
    groups_with_first_index = []
    for key, indices in anagram_map.items():
        first_index = min(indices)
        groups_with_first_index.append((first_index, indices))
    
    # Sort groups by their first appearance index
    groups_with_first_index.sort(key=lambda x: x[0])
    
    # Build the result list
    result = []
    for _, indices in groups_with_first_index:
        # Extract the original words in the order they appeared in the input
        group_words = [words[i] for i in indices]
        result.append(group_words)
    
    return result
