def merge_intervals(intervals: list) -> list:
    if not intervals:
        return []
    
    # Sort by start time to facilitate merging
    sorted_intervals = sorted(intervals, key=lambda x: x[0])
    
    merged = [sorted_intervals[0]]
    
    for current in sorted_intervals[1:]:
        last = merged[-1]
        # Check if the current interval overlaps with the last merged interval
        # They overlap if the start of the current is less than or equal to the end of the last
        if current[0] <= last[1]:
            # Merge them by updating the end of the last interval to be the max of both ends
            last[1] = max(last[1], current[1])
        else:
            # No overlap, add the current interval as a new entry
            merged.append(current)
            
    return merged
