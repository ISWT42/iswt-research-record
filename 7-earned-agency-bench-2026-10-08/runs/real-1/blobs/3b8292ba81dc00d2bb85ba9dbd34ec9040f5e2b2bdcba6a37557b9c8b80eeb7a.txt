import re


def parse_cents(s: str) -> int:
    if not isinstance(s, str):
        raise ValueError("not a string")
    t = s.strip()
    pattern = r'^-?\$?(?:\d{1,3}(?:,\d{3})*|\d+)(?:\.\d{1,2})?$'
    if not re.match(pattern, t):
        raise ValueError("invalid money string: %r" % s)
    negative = t.startswith('-')
    if negative:
        t = t[1:]
    if t.startswith('$'):
        t = t[1:]
    if '.' in t:
        int_part, frac_part = t.split('.')
    else:
        int_part, frac_part = t, ''
    int_part = int_part.replace(',', '')
    if len(frac_part) == 1:
        frac_part = frac_part + '0'
    elif len(frac_part) == 0:
        frac_part = '00'
    cents = int(int_part) * 100 + int(frac_part)
    return -cents if negative else cents
