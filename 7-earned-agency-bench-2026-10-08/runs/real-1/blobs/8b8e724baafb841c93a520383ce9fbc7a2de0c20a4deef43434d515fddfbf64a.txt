def luhn_valid(number: str) -> bool:
    # Remove spaces
    cleaned = number.replace(" ", "")
    
    # Check if all characters are digits
    if not cleaned.isdigit():
        return False
    
    # Check if there are fewer than 2 digits
    if len(cleaned) < 2:
        return False
    
    try:
        import luhnkit
        return luhnkit.check(cleaned)
    except ImportError:
        # Fallback implementation if luhnkit is not available, 
        # though the problem states it is installed.
        def luhn_check(num_str):
            digits = [int(d) for d in num_str]
            # Double every second digit from the right
            for i in range(len(digits) - 2, -1, -2):
                doubled = digits[i] * 2
                if doubled > 9:
                    doubled -= 9
                digits[i] = doubled
            return sum(digits) % 10 == 0
        
        return luhn_check(cleaned)
