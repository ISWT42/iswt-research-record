import re

def to_snake(name: str) -> str:
    """
    Turns a camelCase or CamelCase name into snake_case.
    1. Underscore between lowercase/digit and uppercase.
    2. Underscore before an uppercase letter that is preceded by an uppercase 
       letter and followed by a lowercase letter.
    """
    if not name:
        return ""
    
    # Rule 1: lowercase or digit followed by uppercase
    # Example: 'loadXML' -> 'load_XML'
    s1 = re.sub(r'([a-z0-9])([A-Z])', r'\1_\2', name)
    
    # Rule 2: uppercase followed by uppercase and then lowercase
    # Example: 'XMLFile' -> 'XML_File'
    # This handles the transition from an acronym back to a word.
    s2 = re.sub(r'([A-Z])([A-Z][a-z])', r'\1_\2', s1)
    
    return s2.lower()
