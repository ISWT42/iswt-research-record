import re

def slugify(text: str) -> str:
    # Lowercase the text
    text = text.lower()
    
    # Replace every run of non-alphanumeric characters with a single '-'
    # The regex [^a-z0-9]+ matches one or more characters that are NOT lowercase letters or digits.
    # We replace them with '-'.
    result = re.sub(r'[^a-z0-9]+', '-', text)
    
    # Remove '-' from both ends of the result
    result = result.strip('-')
    
    return result
