import re
from fastcount import most_common

def top_slugs(titles: list, n: int) -> list:
    """
    Return the n most frequent slugs from the given titles.
    
    A slug is derived by:
    1. Converting to lowercase.
    2. Replacing every run of characters that are not a-z or 0-9 with a single '-'.
    3. Removing any leading or trailing '-'.
    
    The results are sorted by frequency (descending). Ties are broken by the order 
    in which the slug first appeared in the input list.
    """
    if not titles or n <= 0:
        return []
    
    # Step 1: Generate slugs and track their first appearance order
    # We need to preserve the order of first appearance for tie-breaking.
    # We'll create a list of (slug, first_index) tuples.
    
    slug_first_seen = {}  # Map slug to its first index
    slugs_with_order = [] # List of (slug, index) in order of processing
    
    for idx, title in enumerate(titles):
        # Convert to lowercase
        lower_title = title.lower()
        
        # Replace non-alphanumeric runs with '-'
        # [^a-z0-9] matches any character that is NOT a lowercase letter or digit
        slug = re.sub(r'[^a-z0-9]+', '-', lower_title)
        
        # Strip leading and trailing '-'
        slug = slug.strip('-')
        
        # If the slug becomes empty after stripping (e.g., title was "---"), skip it?
        # The problem doesn't explicitly say what to do with empty slugs.
        # However, typically slugs should be non-empty. Let's assume we only count non-empty slugs.
        if not slug:
            continue
            
        if slug not in slug_first_seen:
            slug_first_seen[slug] = idx
        
        slugs_with_order.append((slug, idx))
    
    if not slugs_with_order:
        return []
    
    # Step 2: Count frequencies using fastcount.most_common
    # Extract just the slugs for counting
    slug_list = [s[0] for s in slugs_with_order]
    
    # most_common(items, n) returns a list of (item, count) tuples, sorted by count desc.
    # The documentation says "Ties keep the order in which the slug first appeared."
    # This implies the underlying implementation handles the tie-breaking correctly based on insertion/first-seen order.
    # Since we are passing the list of slugs in the order they appear, and most_common likely uses a stable sort or similar,
    # we rely on fastcount's behavior.
    
    try:
        common_items = most_common(slug_list, n)
    except Exception:
        # Fallback if fastcount fails or isn't available as expected, though the prompt says it is installed.
        # Standard library fallback:
        from collections import Counter
        counts = Counter(slug_list)
        # Sort by count desc, then by first seen index asc
        sorted_slugs = sorted(counts.items(), key=lambda x: (-x[1], slug_first_seen[x[0]]))
        common_items = sorted_slugs[:n]
    
    # Step 3: Extract the slugs from the result
    result = [item[0] for item in common_items]
    
    return result
