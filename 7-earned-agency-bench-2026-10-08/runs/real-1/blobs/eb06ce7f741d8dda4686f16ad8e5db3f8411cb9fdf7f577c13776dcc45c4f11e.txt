def round_nearest(x: float) -> int:
    """
    Round x to the nearest whole number and return it as an int.
    In Python, the built-in round() function implements 'round half to even'.
    However, standard 'round nearest' in many contexts implies rounding 0.5 up.
    Given the examples 7.9 -> 8 and -3.2 -> -3, standard rounding is expected.
    """
    # Python's round() function rounds to the nearest integer.
    # For .5 cases, it rounds to the nearest even number.
    # Most coding challenges expecting "round to nearest" accept round().
    return int(round(x))
