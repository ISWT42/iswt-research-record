def moving_average(values: list, k: int) -> list:
    """
    Return the mean of every window of k consecutive values, as floats, in order.
    """
    if k < 1:
        raise ValueError("k must be at least 1")
    
    n = len(values)
    if k > n:
        return []
    
    result = []
    # The number of windows is n - k + 1
    for i in range(n - k + 1):
        window = values[i : i + k]
        window_mean = sum(window) / k
        result.append(float(window_mean))
        
    return result
