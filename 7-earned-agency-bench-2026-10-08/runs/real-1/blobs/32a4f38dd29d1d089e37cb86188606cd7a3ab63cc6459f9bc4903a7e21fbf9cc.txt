import re
from fastcount import most_common

def top_slugs(titles: list, n: int) -> list:
    """
    Returns the n most frequent slugs from a list of titles.
    A slug is lowercase, non-alphanumeric characters are replaced by a single '-',
    and trailing/leading dashes are removed.
    """
    slugs = []
    for title in titles:
        # Convert to lowercase
        slug = title.lower()
        # Replace any sequence of non-alphanumeric characters with a single '-'
        slug = re.sub(r'[^a-z0-9]+', '-', slug)
        # Strip leading and trailing dashes
        slug = slug.strip('-')
        slugs.append(slug)
    
    # fastcount.most_common(items, n) returns the n most frequent items.
    # It preserves the order of first appearance for ties.
    return most_common(slugs, n)
