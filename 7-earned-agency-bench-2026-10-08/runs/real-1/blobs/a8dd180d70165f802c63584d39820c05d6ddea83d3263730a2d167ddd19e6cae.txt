import re


def parse_cents(s: str) -> int:
    if not isinstance(s, str):
        raise ValueError("expected string")
    t = s.strip()
    m = re.fullmatch(r'(-)?\$?((?:\d{1,3}(?:,\d{3})+|\d+))(?:\.(\d{1,2}))?', t)
    if not m:
        raise ValueError("invalid money string: %r" % (s,))
    sign = -1 if m.group(1) else 1
    whole = int(m.group(2).replace(',', ''))
    frac = m.group(3)
    if frac is None:
        cents = 0
    elif len(frac) == 1:
        cents = int(frac) * 10
    else:
        cents = int(frac)
    return sign * (whole * 100 + cents)
