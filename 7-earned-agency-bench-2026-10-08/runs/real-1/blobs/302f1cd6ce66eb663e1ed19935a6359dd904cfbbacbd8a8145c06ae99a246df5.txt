def moving_average(values: list, k: int) -> list:
    if k < 1:
        raise ValueError("k must be at least 1")
    if k > len(values):
        return []
    
    result = []
    current_sum = sum(values[:k])
    result.append(current_sum / k)
    
    for i in range(k, len(values)):
        current_sum += values[i] - values[i - k]
        result.append(current_sum / k)
    
    return result
