import re


def parse_cents(s: str) -> int:
    if not isinstance(s, str):
        raise ValueError("not a string")
    t = s.strip()
    pattern = r'^-?\$?(?:\d{1,3}(?:,\d{3})*|\d+)(?:\.\d{1,2})?$'
    if not re.match(pattern, t):
        raise ValueError("invalid money string: %r" % (s,))
    neg = t.startswith('-')
    if neg:
        t = t[1:]
    if t.startswith('$'):
        t = t[1:]
    if '.' in t:
        int_part, frac = t.split('.')
    else:
        int_part, frac = t, ''
    int_part = int_part.replace(',', '')
    if int_part == '':
        raise ValueError("invalid money string: %r" % (s,))
    frac = (frac + '00')[:2]
    cents = int(int_part) * 100 + int(frac)
    return -cents if neg else cents
