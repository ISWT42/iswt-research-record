import re

def parse_cents(s: str) -> int:
    """
    Parse a money string into whole cents.
    Accepted: optional leading '-', optional '$', digits (thousands commas in groups of 3),
    optional '.' followed by 1 or 2 digits.
    """
    s = s.strip()
    if not s:
        raise ValueError("Empty string")

    # Handle sign
    sign = 1
    if s.startswith('-'):
        sign = -1
        s = s[1:]

    # Handle currency symbol
    if s.startswith('$'):
        s = s[1:]

    if not s:
        raise ValueError("No numeric value provided")

    # Split into whole and fractional parts
    parts = s.split('.')
    if len(parts) > 2:
        raise ValueError("Too many decimal points")

    whole_part = parts[0]
    frac_part = parts[1] if len(parts) > 1 else ""

    # Validate whole part (thousands commas)
    # If there are commas, they must be exactly every 3 digits from the right
    if ',' in whole_part:
        # Check if commas are placed correctly: 1,234,567
        # Regex: digit 1-3, then groups of comma and 3 digits
        if not re.fullmatch(r'\d{1,3}(,\d{3})*', whole_part):
            raise ValueError("Invalid comma placement")
        whole_part = whole_part.replace(',', '')
    else:
        # If no commas, it must be all digits
        if not whole_part.isdigit():
            raise ValueError
