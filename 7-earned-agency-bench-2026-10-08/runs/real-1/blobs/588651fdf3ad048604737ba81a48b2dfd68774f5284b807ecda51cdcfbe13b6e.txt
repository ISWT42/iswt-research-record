import bytepretty


def human_bytes(n: int) -> str:
    if n < 0:
        raise ValueError("negative byte count")
    units = ["B", "KB", "MB", "GB"]
    value = n
    unit_index = 0
    while value >= 1024 and unit_index < len(units) - 1:
        value /= 1024
        unit_index += 1
    if unit_index == 0:
        return bytepretty.format(int(value))
    return bytepretty.format(value)
