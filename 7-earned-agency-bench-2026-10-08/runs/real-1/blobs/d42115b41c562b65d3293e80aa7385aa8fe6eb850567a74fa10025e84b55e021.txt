def roman_to_int(s: str) -> int:
    if not s:
        raise ValueError("Empty string is invalid")
    
    # Check for lowercase or invalid characters
    valid_chars = set('IVXLCDM')
    for char in s:
        if char not in valid_chars:
            raise ValueError(f"Invalid character: {char}")
    
    roman_values = {
        'I': 1,
        'V': 5,
        'X': 10,
        'L': 50,
        'C': 100,
        'D': 500,
        'M': 1000
    }
    
    # Validate the structure of the Roman numeral
    # We need to check that it follows standard form rules
    
    # First, let's parse the value and also validate the format
    # Standard form validation:
    # - I, X, C, M can repeat up to 3 times
    # - V, L, D never repeat
    # - Subtractive pairs only: IV, IX, XL, XC, CD, CM
    
    # Let's do a two-pass approach:
    # 1. Validate the string structure
    # 2. Convert to integer
    
    # Validation logic:
    # Iterate through the string and check validity
    
    i = 0
    n = len(s)
    total = 0
    
    while i < n:
        current_val = roman_values[s[i]]
        
        # Check for repetition constraints
        # Count consecutive same characters
        count = 1
        while i + count < n and s[i + count] == s[i]:
            count += 1
        
        char = s[i]
        
        # V, L, D cannot repeat
        if char in ('V', 'L', 'D'):
            if count > 1:
                raise ValueError(f"Invalid repetition of {char}")
        elif char in ('I', 'X', 'C', 'M'):
            if count > 3:
                raise ValueError(f"Invalid repetition of {char}")
        
        # Check if this is a subtractive pair
        if count == 1 and i + 1 < n:
            next_char = s[i + 1]
            next_val = roman_values[next_char]
            
            if current_val < next_val:
                # This is a subtractive case
                # Valid subtractive pairs: IV, IX, XL, XC, CD, CM
                valid_subtractive = {
                    ('I', 'V'), ('I', 'X'),
                    ('X', 'L'), ('X', 'C'),
                    ('C', 'D'), ('C', 'M')
                }
                
                if (char, next_char) not in valid_subtractive:
                    raise ValueError(f"Invalid subtractive pair: {char}{next_char}")
                
                # Add the subtractive value
                total += next_val - current_val
                i += 2
                continue
            else:
                # Normal additive case
                total += current_val * count
                i += count
                continue
        else:
            # No next character or multiple same characters
            total += current_val * count
            i += count
            continue
    
    return total
