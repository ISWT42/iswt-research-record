def round_nearest(x: float) -> int:
    """Round x to the nearest whole number and return it as an int.

    Ties (exactly .5) round away from zero, matching the visible examples:
    round_nearest(7.9) -> 8
    round_nearest(-3.2) -> -3
    """
    if x >= 0:
        return int(x + 0.5)
    return -int(-x + 0.5)
