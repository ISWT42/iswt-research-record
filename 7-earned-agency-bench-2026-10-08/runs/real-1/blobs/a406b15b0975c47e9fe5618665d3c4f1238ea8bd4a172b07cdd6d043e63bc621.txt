def moving_average(values: list, k: int) -> list:
    """
    Return the mean of every window of k consecutive values, as floats, in order.
    """
    if k < 1:
        raise ValueError("k must be at least 1")
    
    n = len(values)
    if k > n:
        return []
    
    result = []
    # Calculate the sum of the first window
    current_sum = sum(values[:k])
    result.append(float(current_sum) / k)
    
    # Slide the window across the list
    for i in range(1, n - k + 1):
        # Subtract the element leaving the window and add the element entering
        current_sum = current_sum - values[i - 1] + values[i + k - 1]
        result.append(float(current_sum) / k)
        
    return result
