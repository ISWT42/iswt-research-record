import re

def parse_cents(s: str) -> int:
    s = s.strip()
    if not s:
        raise ValueError("Empty string")

    # Regex breakdown:
    # ^(-)?           : Optional leading minus
    # \$?             : Optional dollar sign
    # (               : Start of number group
    #   \d{1,3}       : 1-3 digits
    #   (,\d{3})*     : Zero or more groups of comma + 3 digits
    #   |             : OR
    #   \d+           : Just digits (no commas)
    # )               : End of number group
    # (\.\d{1,2})?    : Optional dot followed by 1 or 2 digits
    # $               : End of string
    pattern = r"^(-)?\$?(\d{1,3}(,\d{3})*|\d+)(\.\d{1,2})?$"
    match = re.match(pattern, s)
    
    if not match:
        raise ValueError(f"Invalid money format: {s}")
    
    sign_group, integer_part, comma_group, decimal_part = match.groups()
    
    # Remove commas from the integer part
    clean_integer = integer_part.replace(',', '')
    
    # Handle decimal part
    if decimal_part:
        # Remove the dot
        fraction = decimal_part[1:]
        # '0.7' -> 70 cents, '0.07' -> 7 cents
        if len(fraction) == 1:
            cents_val = int(fraction) * 10
        else:
            cents_val = int(fraction)
    else:
        cents_val = 0
        
    total_cents = int(clean_integer) * 100 + cents_val
    
    if sign_group == '-':
        total_cents = -total_cents
        
    return total_cents
