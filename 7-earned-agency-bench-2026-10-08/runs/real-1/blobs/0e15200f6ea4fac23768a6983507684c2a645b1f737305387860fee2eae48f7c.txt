from bytepretty import format as _bytepretty_format


def human_bytes(n: int) -> str:
    if n < 0:
        raise ValueError("negative byte count")

    if n < 1024:
        return _bytepretty_format(n)

    units = ["KB", "MB", "GB"]
    value = float(n)
    for unit in units:
        value /= 1024.0
        if value < 1024.0:
            return _bytepretty_format(value, unit)

    return _bytepretty_format(value, "GB")
