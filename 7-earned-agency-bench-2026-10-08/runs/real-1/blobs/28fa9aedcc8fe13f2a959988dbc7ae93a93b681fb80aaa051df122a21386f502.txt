import re
from fastcount import most_common

def top_slugs(titles: list, n: int) -> list:
    """
    Returns the n most frequent slugs from a list of titles.
    A slug is lowercase, non-alphanumeric characters are replaced by a single '-',
    and leading/trailing dashes are removed.
    """
    slugs = []
    for title in titles:
        # Convert to lowercase
        s = title.lower()
        # Replace any run of characters other than a-z and 0-9 with a single '-'
        s = re.sub(r'[^a-z0-9]+', '-', s)
        # Remove leading and trailing dashes
        s = s.strip('-')
        slugs.append(s)
    
    # fastcount.most_common(items, n) returns the n most common elements.
    # According to the prompt, ties should keep the order in which the slug first appeared.
    # most_common typically handles this if the underlying implementation is stable.
    return most_common(slugs, n)
