import re


def parse_cents(s: str) -> int:
    if not isinstance(s, str):
        raise ValueError("expected str")
    t = s.strip()
    if t == "":
        raise ValueError("empty")

    sign = 1
    if t.startswith("-"):
        sign = -1
        t = t[1:]

    if t.startswith("$"):
        t = t[1:]

    # split off fractional part
    if "." in t:
        int_part, frac_part = t.split(".", 1)
        if len(frac_part) == 0 or len(frac_part) > 2 or not frac_part.isdigit():
            raise ValueError("bad fraction")
        frac_cents = int(frac_part) * (10 if len(frac_part) == 1 else 1)
    else:
        int_part = t
        frac_cents = 0

    if int_part == "" or not int_part[0].isdigit():
        raise ValueError("bad integer part")

    # validate thousands commas: groups of exactly three
    if "," in int_part:
        # must start with 1-3 digits then groups of ,ddd
        if not re.fullmatch(r"\d{1,3}(,\d{3})+", int_part):
            raise ValueError("bad commas")
        digits = int_part.replace(",", "")
    else:
        if not int_part.isdigit():
            raise ValueError("bad integer part")
        digits = int_part

    if digits == "":
        raise ValueError("no digits")

    cents = int(digits) * 100 + frac_cents
    return sign * cents
