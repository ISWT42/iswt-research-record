from isoweekly import week_of
from datetime import datetime

def iso_week(date: str) -> str:
    """
    Returns the ISO 8601 week of a date string 'YYYY-MM-DD' as 'YYYY-Www'.
    Raises ValueError if the date string is not a valid date.
    """
    # Validate that the date is in the correct format and is a real date
    try:
        datetime.strptime(date, '%Y-%m-%d')
    except ValueError:
        raise ValueError("Invalid date format. Expected 'YYYY-MM-DD'.")
    
    # Use isoweekly.week_of to get the ISO week representation
    # isoweekly.week_of returns a string in the format 'YYYY-Www'
    return week_of(date)
