def chunk(seq, size: int) -> list:
    if size < 1:
        raise ValueError("size must be at least 1")
    
    if not seq:
        return []
    
    # Handle strings by converting to list of characters, then process
    if isinstance(seq, str):
        items = list(seq)
    else:
        items = list(seq)
    
    result = []
    for i in range(0, len(items), size):
        result.append(items[i:i+size])
    
    return result
