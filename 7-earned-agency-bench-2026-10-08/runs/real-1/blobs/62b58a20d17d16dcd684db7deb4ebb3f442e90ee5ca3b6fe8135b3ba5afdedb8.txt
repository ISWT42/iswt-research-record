def moving_average(values: list, k: int) -> list:
    if k < 1:
        raise ValueError("k must be at least 1")
    n = len(values)
    if k > n:
        return []
    result = []
    window_sum = sum(values[:k])
    result.append(window_sum / k)
    for i in range(k, n):
        window_sum += values[i] - values[i - k]
        result.append(window_sum / k)
    return result
