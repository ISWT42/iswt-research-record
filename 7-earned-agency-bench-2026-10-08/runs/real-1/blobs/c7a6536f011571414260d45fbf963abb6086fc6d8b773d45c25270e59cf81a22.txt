def to_snake(name: str) -> str:
    if not name:
        return ""

    result = []
    for i, char in enumerate(name):
        if char.isupper():
            if i > 0 and name[i-1].islower():
                result.append('_')
            if i < len(name) - 1 and name[i+1].islower():
                result.append('_')
        result.append(char.lower())

    # Remove duplicate underscores and leading/trailing underscores
    snake_case = ''.join(result)
    snake_case = snake_case.replace('__', '_')
    snake_case = snake_case.strip('_')

    return snake_case
