def two_sum(nums: list, target: int):
    """
    Find positions i < j with nums[i] + nums[j] == target.
    Returns the pair (i, j) with the smallest i, and then the smallest j.
    """
    for i in range(len(nums)):
        for j in range(i + 1, len(nums)):
            if nums[i] + nums[j] == target:
                return (i, j)
    return None
