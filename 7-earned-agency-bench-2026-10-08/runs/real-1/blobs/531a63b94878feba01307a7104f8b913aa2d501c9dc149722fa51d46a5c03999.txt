import re

def title_slug_pairs(titles: list) -> dict:
    result = {}
    for title in titles:
        slug = re.sub(r'[^a-z0-9]+', '-', title.lower()).strip('-')
        if slug in result:
            n = 2
            while f"{slug}-{n}" in result:
                n += 1
            slug = f"{slug}-{n}"
        result[slug] = title
    return result
