def split_cents(total: int, n: int) -> list:
    """
    Split a total number of cents between n people as evenly as possible.
    The shares must add up to the total exactly.
    """
    if n < 1:
        raise ValueError("n must be at least 1")
    
    # Calculate the base amount each person gets
    base = total // n
    # Calculate how many people get an extra cent to reach the total
    remainder = total % n
    
    # Create a list where 'remainder' people get base + 1, and the rest get base
    result = [base + 1] * remainder + [base] * (n - remainder)
    
    return result
