def moving_average(values: list, k: int) -> list:
    if k < 1:
        raise ValueError("k must be at least 1")
    if k > len(values):
        return []
    result = []
    for i in range(len(values) - k + 1):
        window = values[i:i + k]
        result.append(sum(window) / k)
    return result
