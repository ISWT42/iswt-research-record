def roman_to_int(s: str) -> int:
    if not s:
        raise ValueError("Empty string is not a valid Roman numeral")

    roman_to_value = {
        'I': 1,
        'V': 5,
        'X': 10,
        'L': 50,
        'C': 100,
        'D': 500,
        'M': 1000
    }

    total = 0
    prev_value = 0

    for char in reversed(s):
        if char not in roman_to_value:
            raise ValueError(f"Invalid Roman numeral character: {char}")

        current_value = roman_to_value[char]

        if current_value < prev_value:
            total -= current_value
        else:
            total += current_value

        prev_value = current_value

    # Validate the Roman numeral format
    if s != to_roman(total):
        raise ValueError("Invalid Roman numeral format")

    return total

def to_roman(num: int) -> str:
    if num < 1 or num > 3999:
        raise ValueError("Number out of range (1-3999)")

    val_to_roman = [
        (1000, 'M'), (900, 'CM'), (500, 'D'), (400, 'CD'),
        (100, 'C'), (90, 'XC'), (50, 'L'), (40, 'XL'),
        (10, 'X'), (9, 'IX'), (5, 'V'), (4, 'IV'), (1, 'I')
    ]

    roman = []
    for value, symbol in val_to_roman:
        while num >= value:
            roman.append(symbol)
            num -= value
    return ''.join(roman)
