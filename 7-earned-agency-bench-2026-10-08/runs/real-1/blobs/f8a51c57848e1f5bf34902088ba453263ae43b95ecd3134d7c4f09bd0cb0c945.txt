import re
from collections import Counter

def top_words(text: str, n: int) -> list:
    if n < 0:
        raise ValueError("n must be non-negative")
    
    # Find all runs of ASCII letters, convert to lowercase
    words = re.findall(r'[a-zA-Z]+', text.lower())
    
    # Count frequencies
    counts = Counter(words)
    
    # Sort by frequency (descending), then by original order of appearance (implicit in Counter.most_common)
    # most_common returns a list of (element, count) tuples sorted by count
    most_common = counts.most_common(n)
    
    return [word for word, count in most_common]
