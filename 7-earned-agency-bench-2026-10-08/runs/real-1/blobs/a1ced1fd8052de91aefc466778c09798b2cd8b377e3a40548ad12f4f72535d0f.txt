def round_nearest(x: float) -> int:
    """
    Round x to the nearest whole number and return it as an int.
    Following the standard rounding behavior where .5 rounds away from zero 
    or to the nearest even, but for basic 'nearest' in most coding contexts 
    without specific tie-breaking rules, the built-in round() is standard.
    However, Python's round() uses 'round half to even'. 
    To ensure standard mathematical rounding (0.5 up), we can use this:
    """
