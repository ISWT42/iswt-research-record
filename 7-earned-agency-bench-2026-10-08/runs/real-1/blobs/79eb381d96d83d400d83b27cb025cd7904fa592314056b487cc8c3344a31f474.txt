def rotate_clockwise(m: list) -> list:
    """
    Rotates a rectangular grid 90 degrees clockwise.
    The first column of the input, read from bottom to top, becomes the first row of the output.
    """
    if not m:
        return []
    
    # If the input is a list containing an empty list, e.g., [[]]
    if not m[0]:
        return []

    # Number of rows and columns in the original matrix
    rows = len(m)
    cols = len(m[0])
    
    # The new matrix will have 'cols' rows and 'rows' columns
    # The element at m[r][c] moves to result[c][(rows - 1) - r]
    result = [[0 for _ in range(rows)] for _ in range(cols)]
    
    for r in range(rows):
        for c in range(cols):
            result[c][(rows - 1) - r] = m[r][c]
            
    return result
