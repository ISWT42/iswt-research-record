import re

def roman_to_int(s: str) -> int:
    if not isinstance(s, str) or not s:
        raise ValueError("Input must be a non-empty string.")
    
    # Standard Roman Numeral Regex
    # M{0,3} : Up to 3000
    # (CM|CD|D?C{0,3}) : 900, 400, or 500-800 / 0-300
    # (XC|XL|L?X{0,3}) : 90, 40, or 50-80 / 0-30
    # (IX|IV|V?I{0,3}) : 9, 4, or 5-8 / 0-3
    pattern = r"^M{0,3}(CM|CD|D?C{0,3})(XC|XL|L?X{0,3})(IX|IV|V?I{0,3})$"
    
    if not re.match(pattern, s):
        raise ValueError("Invalid Roman numeral format.")
    
    # The regex allows empty strings or strings that result in 0, 
    # but the prompt specifies 1 to 3999.
    # Since we already checked 'not s', we just need to ensure it's not empty 
    # (which the regex might match if all groups are empty).
    if s == "":
        raise ValueError("Empty string is not valid.")

    roman_map = {
        'I': 1, 'V': 5, 'X': 10, 'L': 50,
        'C': 100, 'D': 500, 'M': 1000
    }
    
    total = 0
    prev_value = 0
    # Process from right to left to handle subtraction easily
    for char in reversed(s):
        value = roman_map[char]
        if value < prev_value:
            total -= value
        else:
            total += value
        prev_value = value
        
    if total == 0:
        raise ValueError("Value must be between 1 and 3999.")
        
    return total
