def anagram_groups(words: list) -> list:
    """
    Group words that are anagrams of each other, ignoring case.
    
    Args:
        words: A list of strings to group into anagrams.
        
    Returns:
        A list of lists, where each inner list contains words that are anagrams
        of each other. Groups are ordered by the first appearance of any word
        in that group in the input list. Words within a group maintain their
        original order from the input.
    """
    if not words:
        return []
    
    # Dictionary to map sorted lowercase key to list of original words
    anagram_map = {}
    # List to keep track of the order of groups based on first appearance
    group_order = []
    
    for word in words:
        # Create a canonical key for the anagram group (sorted letters, lowercase)
        key = ''.join(sorted(word.lower()))
        
        if key not in anagram_map:
            anagram_map[key] = []
            group_order.append(key)
        
        anagram_map[key].append(word)
    
    # Reconstruct the result list in the order of first appearance
    result = [anagram_map[key] for key in group_order]
    
    return result
