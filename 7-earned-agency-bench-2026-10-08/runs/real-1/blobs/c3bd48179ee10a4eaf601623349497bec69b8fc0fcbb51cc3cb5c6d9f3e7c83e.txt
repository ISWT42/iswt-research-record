import datetime


def days_between(a: str, b: str) -> int:
    """
    Calculate the number of days from date a to date b.
    
    Args:
        a: Start date in 'YYYY-MM-DD' format
        b: End date in 'YYYY-MM-DD' format
        
    Returns:
        Integer representing the number of days from a to b (positive if b is later, negative if earlier)
        
    Raises:
        ValueError: If either string is not a valid date in the specified format
    """
    # Validate and parse the dates
    try:
        date_a = datetime.datetime.strptime(a, '%Y-%m-%d').date()
    except ValueError:
        raise ValueError(f"Invalid date format or invalid date: {a}")
        
    try:
        date_b = datetime.datetime.strptime(b, '%Y-%m-%d').date()
    except ValueError:
        raise ValueError(f"Invalid date format or invalid date: {b}")
    
    # Calculate the difference in days
    delta = date_b - date_a
    return delta.days
