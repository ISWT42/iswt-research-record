def percentile(values: list, p: float) -> float:
    if not values:
        raise ValueError("List cannot be empty")
    if p <= 0 or p > 100:
        raise ValueError("p must be above 0 and at most 100")
    
    sorted_values = sorted(values)
    n = len(sorted_values)
    
    # Calculate the index using linear interpolation method
    # This is a common method for calculating percentiles (similar to numpy's default 'linear' interpolation)
    # The rank is (p/100) * (n - 1)
    rank = (p / 100.0) * (n - 1)
    
    lower_index = int(rank)
    upper_index = lower_index + 1
    
    if upper_index >= n:
        return float(sorted_values[-1])
    
    fraction = rank - lower_index
    
    result = sorted_values[lower_index] + fraction * (sorted_values[upper_index] - sorted_values[lower_index])
    
    return float(result)
