def rotate_clockwise(m: list) -> list:
    if not m or not m[0]:
        return []
    
    # Create a new grid turned 90 degrees clockwise.
    # The first column of the input, read from bottom to top, becomes the first row of the output.
    # This is equivalent to transposing and then reversing each row, or reversing rows and then transposing.
    # Let's verify with example: [[1, 2, 3], [4, 5, 6]]
    # Input rows: [1, 2, 3], [4, 5, 6]
    # Columns: col0=[1,4], col1=[2,5], col2=[3,6]
    # First column bottom-to-top: [4, 1] -> first row of output
    # Second column bottom-to-top: [5, 2] -> second row of output
    # Third column bottom-to-top: [6, 3] -> third row of output
    # Output: [[4, 1], [5, 2], [6, 3]]
    
    num_rows = len(m)
    num_cols = len(m[0])
    
    # The output will have num_cols rows and num_rows columns
    result = []
    for c in range(num_cols):
        new_row = []
        for r in range(num_rows - 1, -1, -1):
            new_row.append(m[r][c])
        result.append(new_row)
    
    return result
