from datetime import date

def age_in_years(born: str, on: str) -> int:
    """
    Calculates the whole number of years a person has completed 
    between the date they were born and a given date.
    """
    # Parse the YYYY-MM-DD strings into date objects
    born_date = date.fromisoformat(born)
    on_date = date.fromisoformat(on)
    
    if on_date < born_date:
        raise ValueError("The 'on' date cannot be before the 'born' date.")
    
    # Calculate the difference in years
    age = on_date.year - born_date.year
    
    # Check if the birthday has occurred yet in the 'on' year
    # If the current month/day is before the birth month/day, subtract one year
    if (on_date.month, on_date.day) < (born_date.month, born_date.day):
        age -= 1
        
    return age
