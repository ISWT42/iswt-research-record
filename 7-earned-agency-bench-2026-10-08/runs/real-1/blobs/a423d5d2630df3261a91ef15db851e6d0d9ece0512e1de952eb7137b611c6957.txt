from datetime import date


def age_in_years(born: str, on: str) -> int:
    b = date.fromisoformat(born)
    o = date.fromisoformat(on)
    if o < b:
        raise ValueError("on is before born")
    years = o.year - b.year
    if (o.month, o.day) < (b.month, b.day):
        years -= 1
    return years
