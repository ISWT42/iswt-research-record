def percentile(values: list, p: float) -> float:
    if not values:
        raise ValueError("values must not be empty")
    if not (0 < p <= 100):
        raise ValueError("p must be in (0, 100]")
    data = sorted(values)
    n = len(data)
    if n == 1:
        return float(data[0])
    # Linear interpolation method (same as numpy's default 'linear')
    k = (n - 1) * (p / 100.0)
    f = int(k)
    c = min(f + 1, n - 1)
    if f == c:
        return float(data[f])
    d = k - f
    return float(data[f] + d * (data[c] - data[f]))
