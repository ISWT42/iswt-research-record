def to_snake(name: str) -> str:
    result = []
    for i, ch in enumerate(name):
        if ch.isupper():
            if i > 0:
                prev = name[i - 1]
                if prev.islower() or prev.isdigit():
                    result.append('_')
                elif prev.isupper() and i + 1 < len(name) and name[i + 1].islower():
                    result.append('_')
            result.append(ch.lower())
        else:
            result.append(ch)
    return ''.join(result)
