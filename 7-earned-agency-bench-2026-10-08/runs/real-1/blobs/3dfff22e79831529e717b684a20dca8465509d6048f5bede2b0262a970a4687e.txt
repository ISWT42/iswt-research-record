def page_summary(count: int, per_page: int) -> list:
    if count < 0 or per_page < 1:
        raise ValueError("Invalid arguments")
    
    if count == 0:
        return []
    
    result = []
    num_pages = (count + per_page - 1) // per_page
    
    for i in range(1, num_pages + 1):
        first = (i - 1) * per_page + 1
        last = min(i * per_page, count)
        result.append(f"{i}: {first}-{last}")
    
    return result
