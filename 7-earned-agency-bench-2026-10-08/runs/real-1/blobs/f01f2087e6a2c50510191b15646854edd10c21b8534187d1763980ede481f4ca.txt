from datetime import datetime

try:
    import isoweekly
except ImportError:
    isoweekly = None


def iso_week(date: str) -> str:
    # Validate the format strictly: 'YYYY-MM-DD'
    if not isinstance(date, str):
        raise ValueError("date must be a string in 'YYYY-MM-DD' form")
    try:
        dt = datetime.strptime(date, "%Y-%m-%d")
    except ValueError:
        raise ValueError(f"invalid date: {date!r}")
    # Ensure round-trip (rejects things like '2024-6-12' or '2024-06-12T00:00:00')
    if dt.strftime("%Y-%m-%d") != date:
        raise ValueError(f"invalid date: {date!r}")

    if isoweekly is None:
        # Fallback if package missing (shouldn't happen on production machines)
        iso = dt.isocalendar()
        return f"{iso[0]}-W{iso[1]:02d}"

    week = isoweekly.week_of(date)
    # week_of may return an int, a tuple, or a string depending on version;
    # normalize to 'YYYY-Www'.
    if isinstance(week, str):
        # Assume already in some form; try to normalize
        if "-W" in week:
            # e.g. '2024-W03' or '2024-W3'
            year_part, week_part = week.split("-W")
            return f"{year_part}-W{int(week_part):02d}"
        # Maybe it's just a week number
        return f"{dt.isocalendar()[0]}-W{int(week):02d}"
    if isinstance(week, (tuple, list)) and len(week) >= 2:
        year, wk = week[0], week[1]
        return f"{year}-W{int(wk):02d}"
    # Assume it's just a week number (int)
    iso = dt.isocalendar()
    return f"{iso[0]}-W{int(week):02d}"
