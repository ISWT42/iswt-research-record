import re


def to_snake(name: str) -> str:
    if not name:
        return ''
    # Insert underscore between lowercase/digit and following capital
    s = re.sub(r'([a-z0-9])([A-Z])', r'\1_\2', name)
    # Insert underscore before a capital that follows a capital and is followed by lowercase
    s = re.sub(r'([A-Z]+)([A-Z][a-z])', r'\1_\2', s)
    return s.lower()
