def anagram_groups(words: list) -> list:
    """
    Group words that are anagrams of each other.
    
    Args:
        words: A list of strings to group into anagrams.
        
    Returns:
        A list of lists, where each inner list contains words that are anagrams
        of each other. Groups are ordered by the first appearance of any word
        in that group in the input list. Words within a group maintain their
        original relative order from the input.
    """
    if not words:
        return []
    
    # Dictionary to store groups of anagrams
    # Key: sorted tuple of characters (canonical form)
    # Value: list of words that match this canonical form
    anagram_map = {}
    
    for word in words:
        # Create a canonical key by sorting the lowercase characters
        key = tuple(sorted(word.lower()))
        
        if key not in anagram_map:
            anagram_map[key] = []
        
        anagram_map[key].append(word)
    
    # We need to preserve the order of groups based on the first occurrence
    # of any word in that group in the original input list.
    # We also need to preserve the order of words within each group as they appeared.
    
    # To determine the order of groups, we find the index of the first word
    # in each group's list within the original 'words' list.
    # Since we iterate through 'words' and append to the map, the first time
    # we encounter a canonical key, the word added is the first one in the input
    # for that group. So we can track the order of keys as they first appear.
    
    ordered_keys = []
    seen_keys = set()
    
    for word in words:
        key = tuple(sorted(word.lower()))
        if key not in seen_keys:
            seen_keys.add(key)
            ordered_keys.append(key)
    
    # Construct the result list using the ordered keys
    result = [anagram_map[key] for key in ordered_keys]
    
    return result
