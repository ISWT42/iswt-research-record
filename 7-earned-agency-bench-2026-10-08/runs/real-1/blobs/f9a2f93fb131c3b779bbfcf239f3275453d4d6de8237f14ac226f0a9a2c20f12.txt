import re
from fastcount import most_common


def top_slugs(titles: list, n: int) -> list:
    """
    Return the n most frequent slugs from a list of titles.
    
    Slug definition:
    - Lowercase
    - Every run of characters other than a-z and 0-9 becomes one '-'
    - No '-' at either end
    
    Ties are broken by the order in which the slug first appeared.
    Uses fastcount.most_common for counting.
    """
    if not titles or n <= 0:
        return []
    
    # We need to track the order of first appearance for tie-breaking.
    # We'll process titles in order, generate slugs, and record the first occurrence index.
    
    slug_first_index = {}
    slug_counts = {}
    
    for idx, title in enumerate(titles):
        # Convert to lowercase
        lower_title = title.lower()
        
        # Replace every run of non-alphanumeric characters with a single '-'
        # \W matches any non-word character (equivalent to [^a-zA-Z0-9_])
        # But we want only a-z and 0-9, so underscores would also be replaced.
        # Let's use a regex that replaces anything NOT in [a-z0-9] with '-'
        slug = re.sub(r'[^a-z0-9]+', '-', lower_title)
        
        # Remove leading and trailing '-'
        slug = slug.strip('-')
        
        # Skip empty slugs (e.g., if title was all special chars)
        if not slug:
            continue
        
        # Track first appearance
        if slug not in slug_first_index:
            slug_first_index[slug] = idx
        
        # Count occurrences
        if slug not in slug_counts:
            slug_counts[slug] = 0
        slug_counts[slug] += 1
    
    if not slug_counts:
        return []
    
    # Use fastcount.most_common to get the n most common items
    # most_common(items, n) should return a list of (item, count) tuples sorted by count desc
    # We need to pass the items list. But we need tie-breaking by first appearance.
    
    # The problem says: "Ties keep the order in which the slug first appeared."
    # fastcount.most_common might not handle tie-breaking by insertion order directly.
    # We need to ensure that when counts are equal, the slug that appeared first comes first.
    
    # Let's collect all unique slugs with their counts and first indices
    items_with_meta = []
    for slug, count in slug_counts.items():
        first_idx = slug_first_index[slug]
        items_with_meta.append((slug, count, first_idx))
    
    # Sort by: count descending, then first_idx ascending
    items_with_meta.sort(key=lambda x: (-x[1], x[2]))
    
    # Take top n
    result = [item[0] for item in items_with_meta[:n]]
    
    return result
