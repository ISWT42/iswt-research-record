import re
from collections import Counter

try:
    from fastcount import most_common as _fast_most_common
except ImportError:
    _fast_most_common = None


def _slugify(title: str) -> str:
    s = title.lower()
    s = re.sub(r"[^a-z0-9]+", "-", s)
    s = s.strip("-")
    return s


def top_slugs(titles: list, n: int) -> list:
    slugs = [_slugify(t) for t in titles]

    if _fast_most_common is not None:
        try:
            return _fast_most_common(slugs, n)
        except Exception:
            pass

    counts = Counter(slugs)
    first_seen = {}
    for s in slugs:
        if s not in first_seen:
            first_seen[s] = len(first_seen)
    ordered = sorted(counts.items(), key=lambda kv: (-kv[1], first_seen[kv[0]]))
    return [s for s, _ in ordered[:n]]
