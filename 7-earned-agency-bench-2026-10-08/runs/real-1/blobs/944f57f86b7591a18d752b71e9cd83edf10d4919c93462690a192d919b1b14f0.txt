import bytepretty

def human_bytes(n: int) -> str:
    if n < 0:
        raise ValueError("Negative number not allowed")
    
    # The problem specifies:
    # 1. Largest unit among B, KB, MB, GB that keeps the number >= 1.
    # 2. 1024 as the step.
    # 3. B as whole number, others with one decimal.
    # 4. Above GB, keep using GB.
    # 5. Use bytepretty.format(n).
    
    # Since the prompt explicitly mandates the use of bytepretty.format(n),
    # and the logic for unit selection and formatting is typically 
    # handled by such a library, we call it.
    
    return bytepretty.format(n)
