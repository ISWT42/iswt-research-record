import re

def title_slug_pairs(titles: list) -> dict:
    result = {}
    for title in titles:
        slug = re.sub(r'[^a-z0-9]+', '-', title.lower()).strip('-')
        if slug in result:
            # ensure uniqueness while keeping all titles
            i = 2
            new_slug = f"{slug}-{i}"
            while new_slug in result:
                i += 1
                new_slug = f"{slug}-{i}"
            slug = new_slug
        result[slug] = title
    return result
