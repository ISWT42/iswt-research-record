import re

def parse_cents(s: str) -> int:
    """
    Parse a money string into whole cents.
    Accepted: optional leading '-', optional '$', digits (thousands commas in groups of 3),
    optional '.' followed by 1 or 2 digits.
    """
    s = s.strip()
    if not s:
        raise ValueError("Empty string")

    # Regex breakdown:
    # ^(-)?             Optional leading minus
    # (\$)?             Optional leading dollar sign
    # (                 Start of number part
    #   \d{1,3}         1-3 digits
    #   (,\d{3})*       Zero or more groups of comma + 3 digits
    #   |               OR
    #   \d+             Just digits (no commas)
    # )
    # (\.(\d{1,2}))?    Optional dot followed by 1 or 2 digits
    # $                 End of string
    
    pattern = r'^(-)?(\$)?((\d{1,3}(,\d{3})*)|(\d+))(\.(\d{1,2}))?$'
    match = re.match(pattern, s)
    
    if not match:
        raise ValueError(f"Invalid money format: {s}")
    
    sign_group = match.group(1)
    # The number part is the combination of the comma-group or the plain-digit group
    # Group 3 is the whole integer part
    integer_part_str = match.group(3).replace(',', '')
    # Group 6 is the fractional part (the digits after the dot)
    fractional_part_str = match.group(7) if match.group(7) is not None else ""
    
    # Convert to cents
    # Integer part * 100
    total_cents = int(integer_part_str) * 100
    
    # Fractional part: 1 digit means 10 cents, 2 digits means 1-99 cents
    if fractional_part_str:
        if len(fractional_part_str) == 1:
            total_cents += int(fractional_part_str) * 10
        else:
            total_cents += int(fractional_part_str)
            
    if sign_group == '-':
        total_cents = -total_cents
        
    return total_cents
