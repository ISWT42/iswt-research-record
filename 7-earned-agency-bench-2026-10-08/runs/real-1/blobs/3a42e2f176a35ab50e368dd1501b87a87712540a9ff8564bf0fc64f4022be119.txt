from datetime import date

def age_in_years(born: str, on: str) -> int:
    """
    Calculates the whole number of years a person has completed between two dates.
    Dates are provided in 'YYYY-MM-DD' format.
    Raises ValueError if 'on' is before 'born'.
    """
    # Parse the date strings into date objects
    b_year, b_month, b_day = map(int, born.split('-'))
    o_year, o_month, o_day = map(int, on.split('-'))
    
    birth_date = date(b_year, b_month, b_day)
    on_date = date(o_year, o_month, o_day)
    
    if on_date < birth_date:
        raise ValueError("The 'on' date cannot be before the 'born' date.")
    
    # Calculate age
    # Initial estimate based on year difference
    age = o_year - b_year
    
    # Adjust if the birthday has not occurred yet in the 'on' year
    # A person has not completed the year if:
    # 1. The current month is before the birth month
    # 2. The current month is the birth month, but the current day is before the birth day
    if (o_month, o_day) < (b_month, b_day):
        age -= 1
        
    return age
