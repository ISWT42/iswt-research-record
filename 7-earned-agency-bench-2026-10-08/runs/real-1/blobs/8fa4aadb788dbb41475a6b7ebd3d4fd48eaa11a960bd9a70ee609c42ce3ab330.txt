import re


def parse_cents(s: str) -> int:
    if not isinstance(s, str):
        raise ValueError("expected string")
    t = s.strip()
    if t == "":
        raise ValueError("empty string")

    negative = False
    if t.startswith("-"):
        negative = True
        t = t[1:]

    if t.startswith("$"):
        t = t[1:]

    # Split off fractional part
    if "." in t:
        if t.count(".") != 1:
            raise ValueError("multiple decimal points")
        int_part, frac_part = t.split(".")
        if not (1 <= len(frac_part) <= 2) or not frac_part.isdigit():
            raise ValueError("bad fractional part")
        if not frac_part.isdigit():
            raise ValueError("bad fractional part")
    else:
        int_part = t
        frac_part = ""

    # Validate integer part with optional thousands commas
    if int_part == "":
        raise ValueError("missing integer part")

    # Must be digits, or groups of 3 separated by commas
    if "," in int_part:
        groups = int_part.split(",")
        # first group 1-3 digits, rest exactly 3
        if not (1 <= len(groups[0]) <= 3 and groups[0].isdigit()):
            raise ValueError("bad thousands grouping")
        for g in groups[1:]:
            if len(g) != 3 or not g.isdigit():
                raise ValueError("bad thousands grouping")
        digits = "".join(groups)
    else:
        if not int_part.isdigit():
            raise ValueError("bad integer part")
        digits = int_part

    if digits == "" or not digits.isdigit():
        raise ValueError("bad integer part")

    dollars = int(digits)
    if frac_part == "":
        cents = 0
    elif len(frac_part) == 1:
        cents = int(frac_part) * 10
    else:
        cents = int(frac_part)

    total = dollars * 100 + cents
    if negative:
        total = -total
    return total
