def compare_versions(a: str, b: str) -> int:
    def parse(v: str):
        parts = v.split('.')
        nums = []
        for p in parts:
            if p == '' or not all(c in '0123456789' for c in p):
                raise ValueError(f"invalid version part: {p!r}")
            nums.append(int(p))
        return nums

    na = parse(a)
    nb = parse(b)
    n = max(len(na), len(nb))
    for i in range(n):
        x = na[i] if i < len(na) else 0
        y = nb[i] if i < len(nb) else 0
        if x < y:
            return -1
        if x > y:
            return 1
    return 0
