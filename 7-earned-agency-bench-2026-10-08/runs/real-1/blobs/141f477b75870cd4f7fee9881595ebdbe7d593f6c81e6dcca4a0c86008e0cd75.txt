def page_summary(count: int, per_page: int) -> list:
    if count < 0 or per_page < 1:
        raise ValueError("count must be non-negative and per_page must be at least 1")
    
    if count == 0:
        return []
    
    summaries = []
    page_num = 1
    for i in range(1, count + 1, per_page):
        first = i
        last = min(i + per_page - 1, count)
        summaries.append(f"{page_num}: {first}-{last}")
        page_num += 1
        
    return summaries
