def parse_cents(s: str) -> int:
    """
    Parse a money string into whole cents.
    
    Accepted format:
    - Optional leading '-'
    - Optional '$'
    - Digits with optional thousands separators (commas in groups of exactly 3)
    - Optional '.' followed by 1 or 2 digits
    
    Spaces around the string are ignored. Anything else raises ValueError.
    """
    # Strip leading and trailing spaces
    s = s.strip()
    
    if not s:
        raise ValueError("Empty string")
    
    # Check for optional leading minus sign
    negative = False
    if s.startswith('-'):
        negative = True
        s = s[1:]
    elif s.startswith('+'):
        # The spec says optional leading '-', doesn't mention '+', so '+' should raise error
        raise ValueError("Unexpected '+' sign")
    
    # Check for optional '$'
    if s.startswith('$'):
        s = s[1:]
    
    if not s:
        raise ValueError("No digits after removing prefix")
    
    # Split on decimal point if present
    if '.' in s:
        parts = s.split('.')
        # There should be exactly one decimal point, so split should give 2 parts
        if len(parts) != 2:
            raise ValueError("Multiple decimal points")
        
        integer_part, fractional_part = parts
        
        # Validate fractional part: must be 1 or 2 digits
        if not fractional_part.isdigit():
            raise ValueError(f"Invalid fractional part: {fractional_part}")
        if len(fractional_part) < 1 or len(fractional_part) > 2:
            raise ValueError(f"Fractional part must be 1 or 2 digits, got {len(fractional_part)}")
        
        # Pad fractional part to 2 digits
        fractional_part = fractional_part.ljust(2, '0')
    else:
        integer_part = s
        fractional_part = '00'
    
    # Validate and clean integer part: remove commas, but validate comma placement
    # Commas are allowed only in groups of exactly three digits
    if ',' in integer_part:
        # Remove commas and check that the original had valid grouping
        # First, let's validate the structure before removing
        # Split by comma
        groups = integer_part.split(',')
        
        # All groups except possibly the first must have exactly 3 digits
        for i, group in enumerate(groups):
            if not group.isdigit():
                raise ValueError(f"Invalid characters in integer part: {group}")
            if i == 0:
                # First group can be 1-3 digits
                if len(group) < 1 or len(group) > 3:
                    raise ValueError(f"First group has invalid length: {len(group)}")
            else:
                # Subsequent groups must have exactly 3 digits
                if len(group) != 3:
                    raise ValueError(f"Group has invalid length: {len(group)}, expected 3")
        
        # Now remove commas
        integer_part = ''.join(groups)
    else:
        # No commas, just digits
        if not integer_part.isdigit():
            raise ValueError(f"Invalid characters in integer part: {integer_part}")
    
    if not integer_part:
        raise ValueError("No digits in integer part")
    
    # Combine integer and fractional parts to get total cents
    # integer_part is the number of dollars
    # fractional_part is already padded to 2 digits representing cents
    total_cents_str = integer_part + fractional_part
    total_cents = int(total_cents_str)
    
    if negative:
        total_cents = -total_cents
    
    return total_cents
