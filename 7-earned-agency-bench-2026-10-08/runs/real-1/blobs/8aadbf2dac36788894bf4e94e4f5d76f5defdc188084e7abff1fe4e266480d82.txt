def rle_encode(s: str) -> str:
    if not s:
        return ""
    
    result = []
    i = 0
    while i < len(s):
        char = s[i]
        count = 0
        while i < len(s) and s[i] == char:
            count += 1
            i += 1
        result.append(f"{count}{char}")
    
    return "".join(result)
