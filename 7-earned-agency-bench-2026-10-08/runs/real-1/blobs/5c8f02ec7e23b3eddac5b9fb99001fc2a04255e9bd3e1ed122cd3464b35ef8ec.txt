def percentile(values: list, p: float) -> float:
    """
    Return the p-th percentile of the numbers in values.
    p is between 0 (exclusive) and 100 (inclusive).
    """
    if not values:
        raise ValueError("List cannot be empty")
    if not (0 < p <= 100):
        raise ValueError("p must be between 0 (exclusive) and 100 (inclusive)")

    sorted_values = sorted(values)
    n = len(sorted_values)
    
    # Using the Nearest Rank method (common for basic percentile implementations)
    # index = ceil((p / 100) * n)
    # Since we want the p-th percentile, we calculate the rank.
    import math
    index = math.ceil((p / 100) * n) - 1
    
    # Ensure index is within bounds (handles p=0 case if it were allowed, 
    # and ensures p=100 returns the last element)
    index = max(0, min(index, n - 1))
    
    return float(sorted_values[index])
