def parse_cents(s: str) -> int:
    s = s.strip()
    if not s:
        raise ValueError

    # Handle optional '-'
    negative = False
    if s.startswith('-'):
        negative = True
        s = s[1:].strip()
        if not s:
            raise ValueError

    # Handle optional '$'
    if s.startswith('$'):
        s = s[1:].strip()
        if not s:
            raise ValueError

    # Split into integer and fractional parts
    if '.' in s:
        integer_part, fractional_part = s.split('.', 1)
    else:
        integer_part, fractional_part = s, ''

    # Validate integer part
    if not integer_part.isdigit():
        if ',' in integer_part:
            parts = integer_part.split(',')
            if len(parts) > 1 and len(parts[-1]) != 3:
                raise ValueError
            integer_part = integer_part.replace(',', '')
            if not integer_part.isdigit():
                raise ValueError
        else:
            raise ValueError

    # Validate fractional part
    if len(fractional_part) > 2:
        raise ValueError
    if fractional_part and not fractional_part.isdigit():
        raise ValueError

    # Combine parts
    cents = int(integer_part) * 100
    if fractional_part:
        cents += int(fractional_part.ljust(2, '0')[:2])

    if negative:
        cents = -cents

    return cents
