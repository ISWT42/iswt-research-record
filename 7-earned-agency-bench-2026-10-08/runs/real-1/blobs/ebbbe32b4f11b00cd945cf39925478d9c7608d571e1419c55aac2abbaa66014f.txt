import datetime

def days_between(a: str, b: str) -> int:
    try:
        date_a = datetime.datetime.strptime(a, '%Y-%m-%d')
        date_b = datetime.datetime.strptime(b, '%Y-%m-%d')
    except ValueError:
        raise ValueError(f"Invalid date format or non-existent date: {a} or {b}")
    
    delta = date_b - date_a
    return delta.days
