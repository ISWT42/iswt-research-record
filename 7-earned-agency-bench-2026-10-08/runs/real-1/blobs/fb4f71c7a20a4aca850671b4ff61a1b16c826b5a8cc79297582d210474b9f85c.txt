try:
    from shortid_tools import encode as _encode
except ImportError:
    _encode = None


def to_base62(n: int) -> str:
    if not isinstance(n, int) or isinstance(n, bool):
        raise ValueError("n must be a non-negative integer")
    if n < 0:
        raise ValueError("n must be non-negative")
    if _encode is None:
        raise ImportError("shortid_tools package is required")
    return _encode(n)
