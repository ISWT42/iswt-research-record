import re

def title_slug_pairs(titles: list) -> dict:
    result = {}
    for title in titles:
        slug = re.sub(r'[^a-z0-9]+', '-', title.lower()).strip('-')
        unique_slug = slug
        counter = 1
        while unique_slug in result:
            unique_slug = f"{slug}-{counter}"
            counter += 1
        result[unique_slug] = title
    return result
