import math

def percentile(values: list, p: float) -> float:
    """
    Return the p-th percentile of the numbers in values.
    p is between 0 (exclusive) and 100 (inclusive).
    """
    if not values:
        raise ValueError("List cannot be empty")
    if p <= 0 or p > 100:
        raise ValueError("p must be between 0 (exclusive) and 100 (inclusive)")

    sorted_values = sorted(values)
    n = len(sorted_values)
    
    # Using the linear interpolation method (commonly used in numpy/excel)
    # index = (p / 100) * (n - 1)
    index = (p / 100) * (n - 1)
    
    lower = math.floor(index)
    upper = math.ceil(index)
    
    if lower == upper:
        return float(sorted_values[int(index)])
    
    # Interpolation
    weight = index - lower
    return float(sorted_values[lower] * (1 - weight) + sorted_values[upper] * weight)
