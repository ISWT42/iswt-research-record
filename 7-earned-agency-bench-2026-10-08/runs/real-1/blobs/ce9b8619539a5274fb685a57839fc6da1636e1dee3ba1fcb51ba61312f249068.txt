from datetime import datetime

try:
    import isoweekly
except ImportError:
    isoweekly = None


def iso_week(date: str) -> str:
    if not isinstance(date, str):
        raise ValueError("date must be a string in 'YYYY-MM-DD' form")
    try:
        d = datetime.strptime(date, "%Y-%m-%d")
    except ValueError:
        raise ValueError(f"invalid date: {date!r}")
    # Ensure round-trip (strptime is lenient about e.g. '2024-6-1')
    if d.strftime("%Y-%m-%d") != date:
        raise ValueError(f"invalid date: {date!r}")

    if isoweekly is not None:
        try:
            week = isoweekly.week_of(date)
        except Exception:
            week = None
        if week is not None:
            return str(week)

    # Fallback: compute ISO week ourselves
    iso = d.isocalendar()
    return f"{iso[0]}-W{iso[1]:02d}"
