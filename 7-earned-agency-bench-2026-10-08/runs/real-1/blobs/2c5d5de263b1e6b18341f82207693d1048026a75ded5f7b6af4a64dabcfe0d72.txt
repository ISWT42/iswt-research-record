from datetime import date

def days_between(a: str, b: str) -> int:
    """
    Returns the number of days between date a and date b.
    Dates must be in 'YYYY-MM-DD' format and be valid calendar dates.
    Raises ValueError if the format is incorrect or the date is invalid.
    """
    def parse_date(date_str):
        # Check for exact length and format (YYYY-MM-DD)
        if len(date_str) != 10:
            raise ValueError("Date must be exactly 10 characters long.")
        if date_str[4] != '-' or date_str[7] != '-':
            raise ValueError("Date must be in YYYY-MM-DD format.")
        
        # Ensure all other characters are digits
        parts = date_str.split('-')
        if len(parts) != 3:
            raise ValueError("Date must be in YYYY-MM-DD format.")
        for part in parts:
            if not part.isdigit():
                raise ValueError("Date components must be numeric.")
        
        # Attempt to create a date object to validate the calendar date
        try:
            return date(int(parts[0]), int(parts[1]), int(parts[2]))
        except ValueError:
            raise ValueError("Invalid calendar date.")

    date_a = parse_date(a)
    date_b = parse_date(b)
    
    delta = date_b - date_a
    return delta.days
