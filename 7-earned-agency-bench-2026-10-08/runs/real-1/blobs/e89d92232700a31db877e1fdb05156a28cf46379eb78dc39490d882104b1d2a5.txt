def percentile(values: list, p: float) -> float:
    if not values:
        raise ValueError("Empty list")
    if p <= 0 or p > 100:
        raise ValueError("p must be above 0 and at most 100")
    
    sorted_values = sorted(values)
    n = len(sorted_values)
    
    # Convert p to a fraction between 0 and 1
    k = (p / 100.0) * (n - 1)
    
    # Linear interpolation
    lower_index = int(k)
    upper_index = lower_index + 1
    
    if upper_index >= n:
        return float(sorted_values[-1])
    
    fraction = k - lower_index
    result = sorted_values[lower_index] + fraction * (sorted_values[upper_index] - sorted_values[lower_index])
    
    return result
