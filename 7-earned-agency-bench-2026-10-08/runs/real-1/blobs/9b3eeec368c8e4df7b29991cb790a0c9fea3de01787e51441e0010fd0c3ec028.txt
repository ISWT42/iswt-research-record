def page_summary(count: int, per_page: int) -> list:
    """
    Returns a list of page summaries for a given count of items and items per page.
    """
    if count < 0:
        raise ValueError("Count cannot be negative.")
    if per_page < 1:
        raise ValueError("per_page must be at least 1.")
    
    if count == 0:
        return []
    
    summaries = []
    # Page numbers start at 1
    # Items are indexed from 1 to count
    page = 1
    current_first = 1
    
    while current_first <= count:
        current_last = min(current_first + per_page - 1, count)
        summaries.append(f"{page}: {current_first}-{current_last}")
        current_first += per_page
        page += 1
        
    return summaries
