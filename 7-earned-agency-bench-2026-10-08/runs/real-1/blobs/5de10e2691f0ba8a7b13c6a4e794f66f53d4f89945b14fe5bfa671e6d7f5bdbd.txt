import bytepretty

def human_bytes(n: int) -> str:
    if n < 0:
        raise ValueError("Byte count cannot be negative")
    
    # The problem specifies using bytepretty.format(n) for formatting.
    # Based on the requirements:
    # - Largest unit among B, KB, MB, GB that keeps the number >= 1.
    # - 1024 as the divisor.
    # - B as whole number, others with one decimal.
    # - Above GB, keep using GB.
    # - Use bytepretty.format(n).
    
    return bytepretty.format(n)
