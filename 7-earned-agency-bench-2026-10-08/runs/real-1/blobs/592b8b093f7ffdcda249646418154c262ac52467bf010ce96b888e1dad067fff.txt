def rle_encode(s: str) -> str:
    if not s:
        return ''
    
    result = []
    current_char = s[0]
    count = 1
    
    for i in range(1, len(s)):
        if s[i] == current_char:
            count += 1
        else:
            result.append(f"{count}{current_char}")
            current_char = s[i]
            count = 1
    
    # Append the last run
    result.append(f"{count}{current_char}")
    
    return ''.join(result)
