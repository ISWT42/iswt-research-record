from datetime import date

def days_between(a: str, b: str) -> int:
    """
    Calculates the number of days between two dates provided in 'YYYY-MM-DD' format.
    Returns a positive integer if b is later than a, negative if b is earlier, 
    and 0 if they are the same day.
    Raises ValueError if the input strings are not valid calendar dates in the 
    specified format.
    """
    def parse_date(date_str):
        # Ensure the string is exactly 10 characters long to avoid partial matches
        if len(date_str) != 10:
            raise ValueError("Date must be in YYYY-MM-DD format")
        
        # Check for the presence of hyphens at the correct positions
        if date_str[4] != '-' or date_str[7] != '-':
            raise ValueError("Date must be in YYYY-MM-DD format")
            
        # Attempt to parse the date; date.fromisoformat handles YYYY-MM-DD 
        # and validates the calendar date (e.g., leap years, days in month)
        try:
            return date.fromisoformat(date_str)
        except ValueError:
            raise ValueError("Invalid calendar date")

    date_a = parse_date(a)
    date_b = parse_date(b)
    
    delta = date_b - date_a
    return delta.days
