def percentile(values: list, p: float) -> float:
    if not values:
        raise ValueError("values must not be empty")
    if not (0 < p <= 100):
        raise ValueError("p must be in (0, 100]")
    data = sorted(values)
    n = len(data)
    if n == 1:
        return float(data[0])
    # Use linear interpolation between closest ranks (numpy's 'linear' method)
    k = (p / 100.0) * (n - 1)
    f = int(k)
    c = min(f + 1, n - 1)
    d = k - f
    return float(data[f] + d * (data[c] - data[f]))
