import re

_PATTERN = re.compile(
    r"""^
    (?P<sign>-)?
    \$?
    (?P<int>\d+)
    (?:,(?P<grp>\d{3}))*
    (?:\.(?P<frac>\d{1,2}))?
    $""",
    re.VERBOSE,
)


def parse_cents(s: str) -> int:
    if not isinstance(s, str):
        raise ValueError("expected a string")
    t = s.strip()
    if t == "":
        raise ValueError("empty string")

    m = _PATTERN.match(t)
    if not m:
        raise ValueError("invalid money string: %r" % (s,))

    int_part = m.group("int")
    grp = m.group("grp")
    frac = m.group("frac")

    # Validate thousands separators: groups of exactly three, only after
    # the first 1-3 digits of the integer part.
    if "," in t:
        # Reconstruct integer portion (without sign/$/frac) and check grouping.
        # The regex already enforces ",ddd" triples; now check the leading group.
        # Find the integer portion between optional sign/$ and optional .frac
        body = t
        if body.startswith("-"):
            body = body[1:]
        if body.startswith("$"):
            body = body[1:]
        if "." in body:
            body = body.split(".", 1)[0]
        # body should be digits and commas
        parts = body.split(",")
        if len(parts) < 2:
            raise ValueError("invalid money string: %r" % (s,))
        first = parts[0]
        if not (1 <= len(first) <= 3):
            raise ValueError("invalid thousands grouping: %r" % (s,))
        for p in parts[1:]:
            if len(p) != 3:
                raise ValueError("invalid thousands grouping: %r" % (s,))
        # Also ensure no empty parts
        if any(p == "" for p in parts):
            raise ValueError("invalid money string: %r" % (s,))

    digits = int_part.replace(",", "")
    cents = int(digits) * 100
    if frac is not None:
        if len(frac) == 1:
            cents += int(frac) * 10
        else:
            cents += int(frac)

    if m.group("sign") == "-":
        cents = -cents
    return cents
