def page_summary(count: int, per_page: int) -> list:
    if count < 0 or per_page < 1:
        raise ValueError("count must be non-negative and per_page must be at least 1")

    if count == 0:
        return []

    pages = []
    for page in range(1, (count // per_page) + 2):
        start = (page - 1) * per_page + 1
        end = min(page * per_page, count)
        if start > count:
            break
        pages.append(f"{page}: {start}-{end}")

    return pages
