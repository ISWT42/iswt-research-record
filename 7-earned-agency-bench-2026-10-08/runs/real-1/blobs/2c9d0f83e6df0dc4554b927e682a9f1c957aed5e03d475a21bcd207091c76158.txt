import re
from collections import Counter

def top_words(text: str, n: int) -> list:
    if n < 0:
        raise ValueError("n must be non-negative")
    
    # Find all runs of ASCII letters and convert to lowercase
    words = re.findall(r'[a-zA-Z]+', text.lower())
    
    # Count frequencies
    counts = Counter(words)
    
    # Sort by frequency (descending), then by original order of appearance (implicit in Counter)
    # The problem asks for the n most frequent. 
    # In Python 3.7+, Counter remembers insertion order.
    # most_common(n) returns the n most common elements.
    # If counts are tied, most_common returns them in the order they were first encountered.
    return [word for word, count in counts.most_common(n)]
