def rotate_clockwise(m: list) -> list:
    if not m:
        return []
    
    # Get dimensions
    num_rows = len(m)
    num_cols = len(m[0])
    
    # The output will have num_cols rows and num_rows columns
    # For a 90 degree clockwise rotation:
    # new_grid[i][j] = old_grid[num_rows - 1 - j][i]
    # Alternatively, the first column of input (bottom to top) becomes first row of output.
    
    result = []
    for c in range(num_cols):
        new_row = []
        for r in range(num_rows - 1, -1, -1):
            new_row.append(m[r][c])
        result.append(new_row)
    
    return result
