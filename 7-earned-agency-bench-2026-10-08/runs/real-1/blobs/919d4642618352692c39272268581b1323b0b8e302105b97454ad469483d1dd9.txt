def flatten(nested: list) -> list:
    """
    Returns a new flat list with all the items of nested, 
    going into inner lists at any depth, in order.
    """
    result = []
    
    def recurse(items):
        for item in items:
            if isinstance(item, list):
                recurse(item)
            else:
                result.append(item)
                
    recurse(nested)
    return result
