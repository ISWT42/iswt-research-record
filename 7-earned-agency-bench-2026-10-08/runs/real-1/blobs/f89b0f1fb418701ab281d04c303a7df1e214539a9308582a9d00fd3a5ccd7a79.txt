def round_nearest(x: float) -> int:
    """
    Round x to the nearest whole number and return it as an int.
    Python's built-in round() uses 'round half to even' strategy.
    For standard mathematical rounding (half up), we can use this approach:
    """
    # The standard behavior for 'round to nearest' in most contexts 
    # is to round .5 away from zero or round half up.
    # However, the examples 7.9 -> 8 and -3.2 -> -3 are handled 
    # correctly by the built-in round() function.
    return int(round(x))
