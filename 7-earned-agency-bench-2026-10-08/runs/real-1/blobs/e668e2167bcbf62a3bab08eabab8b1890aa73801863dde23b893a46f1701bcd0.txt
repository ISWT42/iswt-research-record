def rle_encode(s: str) -> str:
    if not s:
        return ""
    
    res = []
    count = 1
    for i in range(1, len(s)):
        if s[i] == s[i-1]:
            count += 1
        else:
            res.append(f"{count}{s[i-1]}")
            count = 1
    res.append(f"{count}{s[-1]}")
    return "".join(res)
