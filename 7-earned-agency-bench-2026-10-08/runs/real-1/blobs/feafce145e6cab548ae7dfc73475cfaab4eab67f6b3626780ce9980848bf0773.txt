def parse_cents(s: str) -> int:
    s = s.strip()
    if not s:
        raise ValueError

    # Handle optional '-'
    negative = False
    if s[0] == '-':
        negative = True
        s = s[1:].strip()
        if not s:
            raise ValueError

    # Handle optional '$'
    if s[0] == '$':
        s = s[1:].strip()
        if not s:
            raise ValueError

    # Split into integer and fractional parts
    if '.' in s:
        integer_part, fractional_part = s.split('.', 1)
    else:
        integer_part, fractional_part = s, ''

    # Validate and process integer part
    if not integer_part.isdigit():
        # Check for commas
        if ',' not in integer_part:
            raise ValueError
        # Split into groups of 3 digits
        groups = integer_part.split(',')
        # Check each group is 3 digits and no leading/trailing commas
        if len(groups) == 0 or len(groups[-1]) != 3 or any(len(group) != 3 for group in groups[:-1]):
            raise ValueError
        # Join all groups
        integer_part = ''.join(groups)

    # Validate fractional part
    if len(fractional_part) > 2:
        raise ValueError
    if fractional_part and not fractional_part.isdigit():
        raise ValueError

    # Combine parts
    cents = int(integer_part) * 100 + int(fractional_part or 0)

    if negative:
        cents = -cents

    return cents
