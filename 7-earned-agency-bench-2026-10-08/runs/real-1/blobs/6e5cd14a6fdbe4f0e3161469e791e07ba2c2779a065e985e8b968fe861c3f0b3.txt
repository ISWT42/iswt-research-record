def rle_encode(s: str) -> str:
    if not s:
        return ''
    result = []
    i = 0
    n = len(s)
    while i < n:
        j = i
        while j < n and s[j] == s[i]:
            j += 1
        result.append(str(j - i))
        result.append(s[i])
        i = j
    return ''.join(result)
