import re
from fastcount import most_common

def top_slugs(titles: list, n: int) -> list:
    slugs = []
    for title in titles:
        # Lowercase the title
        s = title.lower()
        # Replace any run of characters other than a-z and 0-9 with a single '-'
        s = re.sub(r'[^a-z0-9]+', '-', s)
        # Remove leading and trailing dashes
        s = s.strip('-')
        slugs.append(s)
    
    # most_common returns the n most frequent items.
    # According to the prompt, ties should keep the order in which the slug first appeared.
    # fastcount.most_common is expected to handle the counting and frequency sorting.
    return most_common(slugs, n)
