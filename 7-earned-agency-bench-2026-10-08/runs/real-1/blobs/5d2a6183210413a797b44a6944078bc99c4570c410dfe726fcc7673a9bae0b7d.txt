import re
from fastcount import most_common

def top_slugs(titles: list, n: int) -> list:
    slugs = []
    seen = set()
    for title in titles:
        # Convert to lowercase
        lower_title = title.lower()
        # Replace non-alphanumeric runs with '-'
        slug = re.sub(r'[^a-z0-9]+', '-', lower_title)
        # Remove leading and trailing '-'
        slug = slug.strip('-')
        if slug not in seen:
            seen.add(slug)
            slugs.append(slug)
    # Get the most common slugs
    common_slugs = most_common(slugs, n)
    return [slug for slug, count in common_slugs]
