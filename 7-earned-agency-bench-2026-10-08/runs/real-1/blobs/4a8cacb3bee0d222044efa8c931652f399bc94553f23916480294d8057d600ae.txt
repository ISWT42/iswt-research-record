import re

def top_words(text: str, n: int) -> list:
    if n < 0:
        raise ValueError("n must be non-negative")
    
    # Find all runs of ASCII letters
    words = re.findall(r'[A-Za-z]+', text)
    
    # Convert to lowercase
    words = [w.lower() for w in words]
    
    # Count frequencies
    freq = {}
    for word in words:
        freq[word] = freq.get(word, 0) + 1
    
    # Sort by frequency (descending), then alphabetically for stability (though not strictly required, it's good practice)
    sorted_words = sorted(freq.items(), key=lambda x: (-x[1], x[0]))
    
    # Take top n
    result = [word for word, count in sorted_words[:n]]
    
    return result
