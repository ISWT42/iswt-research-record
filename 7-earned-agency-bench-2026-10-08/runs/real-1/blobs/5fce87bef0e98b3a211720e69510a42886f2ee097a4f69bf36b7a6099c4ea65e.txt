def rotate_clockwise(m: list) -> list:
    if not m:
        return []
    
    # Create a deep copy to ensure the input is not changed
    grid = [row[:] for row in m]
    
    rows = len(grid)
    cols = len(grid[0]) if rows > 0 else 0
    
    if cols == 0:
        return [[] for _ in range(rows)]
    
    # The new grid will have dimensions (cols x rows)
    # New row i corresponds to column (rows - 1 - i) of the original grid? No.
    # Let's trace: 
    # Input: [[1, 2, 3], [4, 5, 6]]
    # Output: [[4, 1], [5, 2], [6, 3]]
    # Output row 0: [4, 1] -> This is col 0 of input, read bottom to top (4 from row 1, 1 from row 0)
    # Output row 1: [5, 2] -> This is col 1 of input, read bottom to top (5 from row 1, 2 from row 0)
    # Output row 2: [6, 3] -> This is col 2 of input, read bottom to top (6 from row 1, 3 from row 0)
    
    new_rows = cols
    new_cols = rows
    result = []
    
    for c in range(cols):
        new_row = []
        for r in range(rows - 1, -1, -1):
            new_row.append(grid[r][c])
        result.append(new_row)
        
    return result
