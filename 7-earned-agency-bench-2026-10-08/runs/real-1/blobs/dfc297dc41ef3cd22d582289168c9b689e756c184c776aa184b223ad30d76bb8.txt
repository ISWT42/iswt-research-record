def title_slug_pairs(titles: list) -> dict:
    """
    Map each title to a slug. If two titles produce the same slug, ensure uniqueness
    by appending a counter suffix (e.g., '-1', '-2') to subsequent duplicates.
    
    The slug is generated by:
    1. Lowercasing the title.
    2. Replacing every run of characters that are not a-z or 0-9 with a single '-'.
    3. Stripping leading and trailing '-' characters.
    
    Returns a dictionary mapping slugs to titles, preserving the order of first appearance.
    """
    import re
    
    def make_slug(title):
        # Convert to lowercase
        s = title.lower()
        # Replace any non-alphanumeric character sequence with a single hyphen
        s = re.sub(r'[^a-z0-9]+', '-', s)
        # Strip leading and trailing hyphens
        s = s.strip('-')
        return s

    result = {}
    seen_slugs = {}  # Maps base slug to count of times it has been used
    
    for title in titles:
        base_slug = make_slug(title)
        
        if base_slug not in seen_slugs:
            # First time seeing this slug
            seen_slugs[base_slug] = 0
            final_slug = base_slug
        else:
            # Duplicate slug found, append counter
            seen_slugs[base_slug] += 1
            final_slug = f"{base_slug}-{seen_slugs[base_slug]}"
            
        result[final_slug] = title
        
    return result
