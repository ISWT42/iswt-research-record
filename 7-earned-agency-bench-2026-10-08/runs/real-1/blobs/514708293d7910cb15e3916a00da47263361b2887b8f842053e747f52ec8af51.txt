import re
from collections import Counter

def top_words(text: str, n: int) -> list:
    if n < 0:
        raise ValueError("n must be non-negative")
    
    # Find all runs of ASCII letters
    words = re.findall(r'[a-zA-Z]+', text)
    # Convert to lowercase
    words = [word.lower() for word in words]
    
    # Count frequencies
    counts = Counter(words)
    
    # Sort by frequency (descending), then by original order of appearance (stable sort)
    # However, the prompt only specifies "most frequent first". 
    # Standard Counter.most_common(n) returns the n most frequent elements.
    # If counts are tied, most_common maintains the order of first encounter.
    
    most_common = counts.most_common(n)
    return [word for word, count in most_common]
