import re
from fastcount import most_common

def top_slugs(titles: list, n: int) -> list:
    """
    Returns the n most frequent slugs from a list of titles.
    A slug is lowercase, non-alphanumeric characters are replaced by a single '-',
    and leading/trailing dashes are removed.
    Ties are handled by the order of first appearance.
    """
    slugs = []
    for title in titles:
        # Convert to lowercase
        s = title.lower()
        # Replace every run of characters other than a-z and 0-9 with one '-'
        s = re.sub(r'[^a-z0-9]+', '-', s)
        # Remove '-' at either end
        s = s.strip('-')
        slugs.append(s)
    
    # fastcount.most_common(items, n) returns the n most frequent items.
    # According to the prompt, ties keep the order in which the slug first appeared.
    # most_common typically preserves this behavior if the underlying counter does.
    return most_common(slugs, n)
