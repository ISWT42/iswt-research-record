import re
from fastcount import most_common

def top_slugs(titles: list, n: int) -> list:
    slugs = []
    for title in titles:
        # Lowercase the title
        s = title.lower()
        # Replace any run of characters other than a-z and 0-9 with a single '-'
        s = re.sub(r'[^a-z0-9]+', '-', s)
        # Strip '-' from either end
        s = s.strip('-')
        slugs.append(s)
    
    # most_common(items, n) returns the n most frequent items.
    # According to the prompt, ties keep the order in which the slug first appeared.
    # fastcount.most_common is expected to handle the counting and the tie-breaking 
    # based on the order of the input list 'slugs'.
    return most_common(slugs, n)
