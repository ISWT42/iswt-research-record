def page_summary(count: int, per_page: int) -> list:
    if count < 0:
        raise ValueError("count must be non-negative")
    if per_page < 1:
        raise ValueError("per_page must be at least 1")

    result = []
    page = 1
    start = 1
    while start <= count:
        end = min(start + per_page - 1, count)
        result.append(f"{page}: {start}-{end}")
        page += 1
        start = end + 1
    return result
