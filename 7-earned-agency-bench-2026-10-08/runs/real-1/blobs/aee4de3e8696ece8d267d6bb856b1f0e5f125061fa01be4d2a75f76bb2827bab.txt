def roman_to_int(s: str) -> int:
    if not isinstance(s, str) or s == "":
        raise ValueError("empty or invalid")
    values = {'I': 1, 'V': 5, 'X': 10, 'L': 50,
              'C': 100, 'D': 500, 'M': 1000}
    subtractive = {'IV', 'IX', 'XL', 'XC', 'CD', 'CM'}
    # Validate structure
    i = 0
    n = len(s)
    total = 0
    while i < n:
        # Check for subtractive pair
        if i + 1 < n and s[i:i+2] in subtractive:
            total += values[s[i+1]] - values[s[i]]
            i += 2
            continue
        # Check for additive symbol
        c = s[i]
        if c not in values:
            raise ValueError("invalid symbol")
        # Count repeats
        j = i
        while j < n and s[j] == c:
            j += 1
        count = j - i
        if c in 'VLD':
            if count != 1:
                raise ValueError("VLD cannot repeat")
        else:
            if count > 3:
                raise ValueError("too many repeats")
        total += values[c] * count
        i = j
    # Validate range
    if total < 1 or total > 3999:
        raise ValueError("out of range")
    # Validate standard form by round-trip
    if _int_to_roman(total) != s:
        raise ValueError("non-standard form")
    return total


def _int_to_roman(num: int) -> str:
    val_map = [
        (1000, 'M'), (900, 'CM'), (500, 'D'), (400, 'CD'),
        (100, 'C'), (90, 'XC'), (50, 'L'), (40, 'XL'),
        (10, 'X'), (9, 'IX'), (5, 'V'), (4, 'IV'), (1, 'I')
    ]
    result = []
    for value, symbol in val_map:
        while num >= value:
            result.append(symbol)
            num -= value
    return ''.join(result)
