import re

def title_slug_pairs(titles: list) -> dict:
    result = {}
    for title in titles:
        slug = re.sub(r'[^a-z0-9]+', '-', title.lower()).strip('-')
        if slug not in result:
            result[slug] = title
        else:
            # Handle collisions by appending a counter to keep slugs unique
            i = 2
            while f"{slug}-{i}" in result:
                i += 1
            result[f"{slug}-{i}"] = title
    return result
