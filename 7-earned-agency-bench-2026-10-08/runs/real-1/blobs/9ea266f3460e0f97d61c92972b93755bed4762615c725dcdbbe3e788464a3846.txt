def rotate_clockwise(m: list) -> list:
    if not m or not m[0]:
        return []
    
    # Create a new grid turned 90 degrees clockwise.
    # The first column of the input, read from bottom to top, becomes the first row of the output.
    # This is equivalent to transposing and then reversing each row, or reversing rows and then transposing.
    # Let's use the standard approach: zip(*reversed(m))
    
    # reversed(m) gives rows from bottom to top
    # zip(*...) unpacks the rows and zips them together, effectively taking columns
    
    rotated = [list(row) for row in zip(*reversed(m))]
    
    return rotated
