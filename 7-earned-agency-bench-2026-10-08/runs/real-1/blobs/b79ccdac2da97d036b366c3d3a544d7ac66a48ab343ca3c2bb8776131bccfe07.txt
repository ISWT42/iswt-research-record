import re

def roman_to_int(s: str) -> int:
    if not isinstance(s, str) or s == "":
        raise ValueError("Input must be a non-empty string.")
    
    # Standard Roman Numeral Regex for 1-3999
    # M{0,3} : Thousands (0-3000)
    # (CM|CD|D?C{0,3}) : Hundreds (900, 400, 500-800, 0-300)
    # (XC|XL|L?X{0,3}) : Tens (90, 40, 50-80, 0-30)
    # (IX|IV|V?I{0,3}) : Ones (9, 4, 5-8, 0-3)
    pattern = r"^M{0,3}(CM|CD|D?C{0,3})(XC|XL|L?X{0,3})(IX|IV|V?I{0,3})$"
    
    if not re.match(pattern, s):
        raise ValueError("Invalid Roman numeral standard form.")
    
    # The regex allows empty strings (all groups optional), but we already checked s == ""
    # However, the regex could match a string that is technically empty if we didn't check.
    # The problem states 1 to 3999, so we must ensure it's not just an empty match.
    # Since we checked s == "", and the regex matches the whole string, we are safe.

    roman_map = {
        'I': 1, 'V': 5, 'X': 10, 'L': 50,
        'C': 100, 'D': 500, 'M': 1000
    }
    
    total = 0
    prev_value = 0
    
    # Process from right to left to handle subtraction easily
    for char in reversed(s):
        value = roman_map[char]
        if value < prev_value:
            total -= value
        else:
            total += value
        prev_value = value
        
    if total == 0:
        raise ValueError("Roman numeral must represent a value between 1 and 3999.")
        
    return total
