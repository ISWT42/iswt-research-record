def roman_to_int(s: str) -> int:
    if not s:
        raise ValueError("Empty string is invalid")
    
    # Check for lowercase or invalid characters
    valid_chars = set('IVXLCDM')
    for char in s:
        if char not in valid_chars:
            raise ValueError(f"Invalid character: {char}")
    
    roman_values = {
        'I': 1,
        'V': 5,
        'X': 10,
        'L': 50,
        'C': 100,
        'D': 500,
        'M': 1000
    }
    
    # Validate the structure of the Roman numeral
    # We need to ensure it's in standard form.
    # A simple way is to convert back to Roman and check equality, but that's complex.
    # Instead, we can parse it and check for validity during parsing.
    
    # Let's use a state machine or regex-like validation.
    # Standard Roman numerals follow specific patterns.
    # Thousands: M{0,3}
    # Hundreds: (CM|CD|D?C{0,3})
    # Tens: (XC|XL|L?X{0,3})
    # Ones: (IX|IV|V?I{0,3})
    
    import re
    
    pattern = r'^M{0,3}(CM|CD|D?C{0,3})(XC|XL|L?X{0,3})(IX|IV|V?I{0,3})$'
    if not re.match(pattern, s):
        raise ValueError("Invalid Roman numeral format")
    
    total = 0
    prev_value = 0
    
    for char in reversed(s):
        value = roman_values[char]
        if value < prev_value:
            total -= value
        else:
            total += value
        prev_value = value
        
    return total
