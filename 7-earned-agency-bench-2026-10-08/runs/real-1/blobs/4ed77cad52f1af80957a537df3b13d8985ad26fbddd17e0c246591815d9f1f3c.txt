import re

def title_slug_pairs(titles: list) -> dict:
    """
    Converts titles to slugs and maps them to the original titles.
    If slugs collide, they are made unique by appending a numeric suffix.
    """
    result = {}
    for title in titles:
        # Lowercase the title
        slug = title.lower()
        # Replace any run of non-alphanumeric characters with a single '-'
        slug = re.sub(r'[^a-z0-9]+', '-', slug)
        # Strip leading and trailing '-'
        slug = slug.strip('-')
        
        # Handle collisions
        if slug in result:
            base_slug = slug
            counter = 1
            while f"{base_slug}-{counter}" in result:
                counter += 1
            slug = f"{base_slug}-{counter}"
            
        result[slug] = title
        
    return result
