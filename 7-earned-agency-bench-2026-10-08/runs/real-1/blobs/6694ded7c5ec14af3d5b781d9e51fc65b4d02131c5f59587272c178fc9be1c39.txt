def parse_cents(s: str) -> int:
    s = s.strip()
    if not s:
        raise ValueError

    # Handle optional '-'
    negative = False
    if s[0] == '-':
        negative = True
        s = s[1:].strip()
        if not s:
            raise ValueError

    # Handle optional '$'
    if s[0] == '$':
        s = s[1:].strip()
        if not s:
            raise ValueError

    # Split into integer and fractional parts
    if '.' in s:
        integer_part, fractional_part = s.split('.', 1)
    else:
        integer_part, fractional_part = s, ''

    # Validate integer part
    if not integer_part.isdigit():
        # Check for commas
        if ',' not in integer_part:
            raise ValueError
        parts = integer_part.split(',')
        for part in parts:
            if not part.isdigit():
                raise ValueError
            if len(part) != 3:
                raise ValueError
        # Reconstruct the number without commas
        integer_part = ''.join(parts)

    # Validate fractional part
    if len(fractional_part) not in (0, 1, 2):
        raise ValueError
    if not fractional_part.isdigit():
        raise ValueError

    # Combine parts
    cents = int(integer_part + fractional_part.ljust(2, '0'))

    return -cents if negative else cents
