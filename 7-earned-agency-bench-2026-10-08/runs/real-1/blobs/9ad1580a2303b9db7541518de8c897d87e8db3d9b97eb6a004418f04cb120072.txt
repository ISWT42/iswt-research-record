import re

def parse_cents(s: str) -> int:
    s = s.strip()
    if not s:
        raise ValueError("Empty string")
    
    # Regex breakdown:
    # ^(-)?         Optional leading minus
    # (\$)?         Optional leading dollar sign
    # (             Group for the numeric part:
    #   \d{1,3}(,\d{3})*  Thousands commas: 1-3 digits followed by zero or more groups of ,XXX
    #   |                OR
    #   \d+              Just digits
    # )
    # (\.\d{1,2})?  Optional decimal point followed by 1 or 2 digits
    # $
    pattern = r'^([-]?)([\$]?)((\d{1,3}(,\d{3})*)|(\d+))(\.\d{1,2})?$'
    match = re.match(pattern, s)
    
    if not match:
        raise ValueError("Invalid money format")
    
    sign_group, dollar_group, numeric_part, comma_group, digit_group, decimal_group = match.groups()
    
    # The regex groups are a bit messy due to the OR for commas. 
    # We need the actual digits of the whole number part.
    whole_str = numeric_part.replace(',', '')
    
    # Handle decimal part
    cents_str = whole_str + '00'
    if decimal_group:
        dec_val = decimal_group[1:] # remove the '.'
        if len(dec_val) == 1:
            dec_val += '0'
        cents_str = whole_str + dec_val
    
    try:
        result = int(cents_str)
        if sign_group == '-':
            result = -result
        return result
    except ValueError:
        raise ValueError("Invalid numeric value")
