def rotate_clockwise(m: list) -> list:
    if not m or not m[0]:
        return []
    
    # Get dimensions
    rows = len(m)
    cols = len(m[0])
    
    # The new grid will have 'cols' rows and 'rows' columns
    # For a 90 degree clockwise rotation:
    # new_grid[i][j] = m[rows - 1 - j][i]
    # Alternatively, we can construct it by taking columns from bottom to top
    
    new_grid = []
    for c in range(cols):
        new_row = []
        for r in range(rows - 1, -1, -1):
            new_row.append(m[r][c])
        new_grid.append(new_row)
    
    return new_grid
