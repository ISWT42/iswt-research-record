def title_slug_pairs(titles: list) -> dict:
    result = {}
    for title in titles:
        slug = []
        prev_dash = False
        for ch in title.lower():
            if ch.isalnum() and ch.isascii():
                slug.append(ch)
                prev_dash = False
            else:
                if not prev_dash:
                    slug.append('-')
                    prev_dash = True
        # strip leading/trailing dashes
        s = ''.join(slug).strip('-')
        # ensure uniqueness
        if s in result:
            i = 2
            while f"{s}-{i}" in result:
                i += 1
            s = f"{s}-{i}"
        result[s] = title
    return result
