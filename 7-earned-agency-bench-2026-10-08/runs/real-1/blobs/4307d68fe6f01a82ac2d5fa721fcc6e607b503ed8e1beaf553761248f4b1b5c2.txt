def parse_cents(s: str) -> int:
    s = s.strip()
    if not s:
        raise ValueError

    negative = False
    if s[0] == '-':
        negative = True
        s = s[1:].strip()
        if not s:
            raise ValueError

    if s[0] == '$':
        s = s[1:].strip()
        if not s:
            raise ValueError

    # Split into integer and fractional parts
    parts = s.split('.')
    if len(parts) > 2:
        raise ValueError
    integer_part = parts[0]
    fractional_part = parts[1] if len(parts) == 2 else ''

    # Validate integer part
    if not integer_part:
        raise ValueError
    for i in range(len(integer_part)):
        if integer_part[i] == ',':
            if (len(integer_part) - i - 1) % 3 != 2:
                raise ValueError
        elif not integer_part[i].isdigit():
            raise ValueError

    # Remove commas from integer part
    integer_part = integer_part.replace(',', '')

    # Validate fractional part
    if fractional_part:
        if len(fractional_part) not in (1, 2):
            raise ValueError
        if not fractional_part.isdigit():
            raise ValueError

    # Combine parts
    cents = int(integer_part) * 100
    if fractional_part:
        cents += int(fractional_part)

    if negative:
        cents = -cents

    return cents
