import re
from collections import Counter

def top_words(text: str, n: int) -> list:
    if n < 0:
        raise ValueError("n must be non-negative")
    
    # Find all runs of ASCII letters, convert to lowercase
    words = re.findall(r'[a-zA-Z]+', text.lower())
    
    # Count frequencies
    counts = Counter(words)
    
    # Sort by frequency (descending), then by first appearance (implicit in Counter.most_common)
    # The prompt asks for the n most frequent words. 
    # most_common(n) returns the n most frequent elements.
    # In case of ties, most_common maintains the order of first encounter.
    return [word for word, count in counts.most_common(n)]
