import re

def parse_cents(s: str) -> int:
    s = s.strip()
    if not s:
        raise ValueError("Empty string")

    # Handle optional leading minus
    sign = 1
    if s.startswith('-'):
        sign = -1
        s = s[1:]

    # Handle optional leading dollar sign
    if s.startswith('$'):
        s = s[1:]

    if not s:
        raise ValueError("No numeric value found")

    # Split into whole and fractional parts
    parts = s.split('.')
    if len(parts) > 2:
        raise ValueError("Too many decimal points")

    whole_part = parts[0]
    frac_part = parts[1] if len(parts) > 1 else ""

    # Validate whole part (thousands commas)
    # Must be digits or digits with commas every 3 places
    if not whole_part:
        # If there's no whole part, it must be like '.50'
        # But the spec says "digits (thousands commas...)", 
        # usually implying at least one digit if no decimal, 
        # but let's check if '.5' is allowed.
        # "then digits... then an optional '.'" suggests digits are required.
        # However, common parsing allows '.5'. Let's re-read:
        # "then digits... then an optional '.' followed by 1 or 2 digits"
        # This implies digits MUST precede the optional dot.
        raise ValueError("Whole part missing")

    # Check for commas
    if ',' in whole_part:
        # First group 1-3 digits, then groups of exactly 3 digits
        # Regex: ^\d{1,3}(,\d{3})+$
        if not re.match(r'^\d{1,3}(,\d{3})+$', whole_part):
            raise ValueError("Invalid comma placement")
        whole_val = int(whole_part.replace(',', ''))
    else:
        # No commas, just digits
        if not whole_part.isdigit():
            raise ValueError("Invalid characters in whole part")
        whole_val = int(whole_part)

    # Validate fractional part
    if frac_part:
        if not (1 <= len(frac_part) <= 2 and frac_part.isdigit()):
            raise ValueError("Invalid fractional part")
        
        if len(frac_part) == 1:
            frac_val = int(frac_part) * 10
        else:
            frac_val = int(frac_part)
    else:
        frac_val = 0

    return sign * (whole_val * 100 + frac_val)
