import re

def roman_to_int(s: str) -> int:
    if not isinstance(s, str):
        raise ValueError("Input must be a string")
    
    # Standard Roman Numeral Regex for 1-3999
    # M{0,3} : Thousands (0-3000)
    # (CM|CD|D?C{0,3}) : Hundreds (900, 400, or 500-800, 0-300)
    # (XC|XL|L?X{0,3}) : Tens (90, 40, or 50-80, 0-30)
    # (IX|IV|V?I{0,3}) : Ones (9, 4, or 5-8, 0-3)
    pattern = r'^M{0,3}(CM|CD|D?C{0,3})(XC|XL|L?X{0,3})(IX|IV|V?I{0,3})$'
    
    if not s:
        raise ValueError("Empty string")
    
    # Check for lowercase or invalid characters via regex match
    # The regex ensures only valid symbols and standard form
    match = re.match(pattern, s)
    if not match:
        raise ValueError("Invalid Roman numeral or non-standard form")
    
    # The regex allows empty string (all groups optional), but we handled that above.
    # However, the regex could match an empty string if s was empty.
    # Since we checked 'if not s', we are safe.
    
    roman_map = {
        'I': 1, 'V': 5, 'X': 10, 'L': 50,
        'C': 100, 'D': 500, 'M': 1000
    }
    
    total = 0
    prev_value = 0
    
    # Process from right to left to handle subtraction easily
    for char in reversed(s):
        value = roman_map[char]
        if value < prev_value:
            total -= value
        else:
            total += value
        prev_value = value
        
    # The problem specifies 1 to 3999. 
    # The regex allows 0 (empty string), but we caught that.
    # If the regex matched but total is 0, it's invalid.
    if total == 0:
        raise ValueError("Value must be between 1 and 3999")
        
    return total
