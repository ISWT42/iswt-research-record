import re

def slugify(text: str) -> str:
    # Convert to lowercase
    text = text.lower()
    # Replace any non-alphanumeric character with a hyphen
    text = re.sub(r'[^a-z0-9]+', '-', text)
    # Remove hyphens from both ends
    text = text.strip('-')
    return text
