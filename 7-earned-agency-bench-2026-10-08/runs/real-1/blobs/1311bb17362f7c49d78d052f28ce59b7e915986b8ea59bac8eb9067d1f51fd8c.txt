import re

def roman_to_int(s: str) -> int:
    if not isinstance(s, str):
        raise ValueError("Input must be a string")
    
    # Standard Roman Numeral Regex for 1-3999
    # M{0,3} : Thousands (0-3000)
    # (CM|CD|D?C{0,3}) : Hundreds (900, 400, or 500-800 / 0-300)
    # (XC|XL|L?X{0,3}) : Tens (90, 40, or 50-80 / 0-30)
    # (IX|IV|V?I{0,3}) : Ones (9, 4, or 5-8 / 0-3)
    pattern = r"^M{0,3}(CM|CD|D?C{0,3})(XC|XL|L?X{0,3})(IX|IV|V?I{0,3})$"
    
    if not re.match(pattern, s):
        raise ValueError("Invalid Roman numeral")
    
    if s == "":
        raise ValueError("Empty string is not a valid Roman numeral")

    roman_map = {
        'I': 1, 'V': 5, 'X': 10, 'L': 50,
        'C': 100, 'D': 500, 'M': 1000
    }
    
    total = 0
    prev_val = 0
    # Process from right to left to handle subtraction easily
    for char in reversed(s):
        val = roman_map[char]
        if val >= prev_val:
            total += val
        else:
            total -= val
        prev_val = val
        
    return total
