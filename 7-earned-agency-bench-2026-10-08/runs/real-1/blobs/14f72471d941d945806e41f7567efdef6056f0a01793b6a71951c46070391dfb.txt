def split_cents(total: int, n: int) -> list:
    """
    Split a total number of cents between n people as evenly as possible.
    The shares must add up to the total exactly.
    """
    if n < 1:
        raise ValueError("n must be at least 1")
    
    base_share = total // n
    remainder = total % n
    
    # Create a list where 'remainder' people get base_share + 1
    # and the rest get base_share.
    result = [base_share + 1] * remainder + [base_share] * (n - remainder)
    
    return result
