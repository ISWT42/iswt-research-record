def compare_versions(a: str, b: str) -> int:
    def parse_version(v):
        parts = v.split('.')
        parsed = []
        for p in parts:
            if not p or not p.isdigit():
                raise ValueError("Version parts must be non-empty and consist only of digits")
            parsed.append(int(p))
        return parsed

    parts_a = parse_version(a)
    parts_b = parse_version(b)
    
    max_len = max(len(parts_a), len(parts_b))
    
    for i in range(max_len):
        val_a = parts_a[i] if i < len(parts_a) else 0
        val_b = parts_b[i] if i < len(parts_b) else 0
        
        if val_a > val_b:
            return 1
        if val_a < val_b:
            return -1
            
    return 0
