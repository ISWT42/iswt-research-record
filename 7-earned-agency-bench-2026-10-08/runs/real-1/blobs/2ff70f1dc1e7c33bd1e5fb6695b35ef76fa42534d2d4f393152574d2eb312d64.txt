def anagram_groups(words: list) -> list:
    """
    Group words that are anagrams of each other, ignoring case.
    Returns a list of groups, where each group is a list of original words.
    Groups are ordered by the first appearance of their members in the input.
    """
    groups_map = {}
    order = []
    
    for word in words:
        # Normalize word for anagram checking: lowercase and sorted characters
        key = "".join(sorted(word.lower()))
        
        if key not in groups_map:
            groups_map[key] = []
            order.append(key)
        
        groups_map[key].append(word)
    
    # Return the groups in the order their first member appeared
    return [groups_map[key] for key in order]
