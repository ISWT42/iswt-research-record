import re

def parse_cents(s: str) -> int:
    """
    Parse a money string into whole cents.
    Accepted: optional leading '-', optional '$', digits (thousands commas in groups of 3),
    optional '.' followed by 1 or 2 digits.
    """
    s = s.strip()
    if not s:
        raise ValueError("Empty string")

    # Regex breakdown:
    # ^(-)?            : Optional leading minus
    # \$?              : Optional dollar sign
    # (                : Start of number part
    #   \d{1,3}(,\d{3})* : Digits with commas every 3 places
    #   | \d+          : Or just digits without commas
    # )
    # (\.(\d{1,2}))?   : Optional dot followed by 1 or 2 digits
    # $                : End of string
    
    # To strictly enforce "thousands commas are allowed only in groups of exactly three",
    # we must ensure that if a comma is present, the pattern follows the comma rule.
    # If no comma is present, it's just digits.
    
    pattern = r'^(-)?\$?(\d{1,3}(?:,\d{3})*|\d+)(\.(\d{1,2}))?$'
    match = re.match(pattern, s)
    
    if not match:
        raise ValueError(f"Invalid money format: {s}")
