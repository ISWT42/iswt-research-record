import re

def roman_to_int(s: str) -> int:
    if not isinstance(s, str) or not s:
        raise ValueError("Input must be a non-empty string.")
    
    # Regular expression to validate standard Roman numeral form (1-3999)
    # M{0,3} : Thousands (0-3000)
    # (CM|CD|D?C{0,3}) : Hundreds (900, 400, or 500-800 / 0-300)
    # (XC|XL|L?X{0,3}) : Tens (90, 40, or 50-80 / 0-30)
    # (IX|IV|V?I{0,3}) : Ones (9, 4, or 5-8 / 0-3)
    pattern = r"^M{0,3}(CM|CD|D?C{0,3})(XC|XL|L?X{0,3})(IX|IV|V?I{0,3})$"
    
    if not re.match(pattern, s):
        raise ValueError("Invalid Roman numeral format.")
    
    # The regex allows empty strings, but the problem says empty string raises ValueError
    if s == "":
        raise ValueError("Empty string is not allowed.")

    roman_map = {
        'I': 1, 'V': 5, 'X': 10, 'L': 50,
        'C': 100, 'D': 500, 'M': 1000
    }
    
    total = 0
    prev_value = 0
    # Process from right to left to handle subtraction easily
    for char in reversed(s):
        value = roman_map[char]
        if value < prev_value:
            total -= value
        else:
            total += value
        prev_value = value
        
    # The problem specifies 1 to 3999. 
    # The regex handles the upper bound and the empty string check handles the lower.
    if total == 0:
        raise ValueError("Roman numeral must represent a value between 1 and 3999.")
        
    return total
