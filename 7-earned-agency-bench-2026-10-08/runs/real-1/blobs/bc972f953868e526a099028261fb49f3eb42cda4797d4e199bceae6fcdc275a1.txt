def merge_intervals(intervals: list) -> list:
    if not intervals:
        return []

    # Create a sorted copy of the intervals to avoid mutating the original list
    sorted_intervals = sorted(intervals, key=lambda x: x[0])
    
    merged = []
    if not sorted_intervals:
        return merged

    # Initialize with the first interval
    current_start, current_end = sorted_intervals[0]

    for i in range(1, len(sorted_intervals)):
        next_start, next_end = sorted_intervals[i]
        
        # If the next interval starts before or exactly when the current one ends,
        # they share at least one number and should be merged.
        if next_start <= current_end:
            current_end = max(current_end, next_end)
        else:
            # No overlap, push the completed interval and start a new one
            merged.append([current_start, current_end])
            current_start, current_end = next_start, next_end

    # Append the final interval
    merged.append([current_start, current_end])
    
    return merged
