import re

def parse_cents(s: str) -> int:
    s = s.strip()
    if not s:
        raise ValueError("Empty string")

    # Regex breakdown:
    # ^(-?)             Optional leading minus
    # (\$?)             Optional leading dollar sign
    # (                 Group for the numeric part:
    #   \d{1,3}(,\d{3})*  1-3 digits followed by zero or more groups of comma+3 digits
    #   |                OR
    #   \d+               Just digits (no commas)
    # )
    # (\.\d{1,2})?      Optional dot followed by 1 or 2 digits
    # $                 End of string
    
    pattern = r'^(-?)(\$?)(\d{1,3}(?:,\d{3})*|\d+)(\.\d{1,2})?$'
    match = re.match(pattern, s)
    
    if not match:
        raise ValueError(f"Invalid money format: {s}")
    
    sign_str, dollar_str, integer_part, decimal_part = match.groups()
    
    # Remove commas from integer part
    integer_part = integer_part.replace(',', '')
    
    # Handle decimal part
    if decimal_part:
        # Remove the dot
        decimal_val = decimal_part[1:]
        # '0.7' -> 70 cents, '0.07' -> 7 cents
        if len(decimal_val) == 1:
            cents_str = decimal_val + '0'
        else:
            cents_str = decimal_val
    else:
        cents_str = '00'
    
    total_cents = int(integer_part) * 100 + int(cents_str)
    
    if sign_str == '-':
        total_cents = -total_cents
        
    return total_cents
