def parse_cents(s: str) -> int:
    """
    Parse a money string into whole cents.
    
    Accepted format:
    - Optional leading '-'
    - Optional '$'
    - Digits with optional thousands separators (commas in groups of exactly 3)
    - Optional '.' followed by 1 or 2 digits
    
    Spaces around the string are ignored.
    Anything else raises ValueError.
    """
    # Strip whitespace
    s = s.strip()
    
    if not s:
        raise ValueError("Empty string")
    
    # Check for negative sign
    negative = False
    if s.startswith('-'):
        negative = True
        s = s[1:]
    elif s.startswith('+'):
        # The spec says "optional leading '-'" but doesn't mention '+'. 
        # However, typically '+' is not allowed unless specified. Let's be strict.
        # Actually, re-reading: "an optional leading '-', then an optional '$'". 
        # It does NOT mention '+'. So '+' should raise ValueError.
        raise ValueError("Unexpected character '+'")
    
    # Check for optional '$'
    if s.startswith('$'):
        s = s[1:]
    
    if not s:
        raise ValueError("No digits after optional $")
    
    # Split by decimal point
    if '.' in s:
        parts = s.split('.')
        if len(parts) != 2:
            raise ValueError("Multiple decimal points")
        integer_part, fractional_part = parts
        
        # Validate fractional part: must be 1 or 2 digits
        if not fractional_part.isdigit():
            raise ValueError(f"Invalid fractional part: {fractional_part}")
        if len(fractional_part) < 1 or len(fractional_part) > 2:
            raise ValueError(f"Fractional part must be 1 or 2 digits, got {len(fractional_part)}")
        
        # Pad fractional part to 2 digits
        if len(fractional_part) == 1:
            fractional_part += '0'
    else:
        integer_part = s
        fractional_part = '00'
    
    # Validate and clean integer part
    # Remove commas and validate format
    if ',' in integer_part:
        # Check that commas are in groups of exactly 3
        # First, remove all commas
        no_comma = integer_part.replace(',', '')
        
        # Now check the original string for valid comma placement
        # Commas should appear only in groups of 3 from the right
        # We can do this by checking segments between commas
        segments = integer_part.split(',')
        
        # There must be at least one segment
        if not segments:
            raise ValueError("Invalid comma placement")
        
        # All segments except possibly the first must be exactly 3 digits
        for i, seg in enumerate(segments):
            if not seg.isdigit():
                raise ValueError(f"Non-digit in integer part: {seg}")
            if i == 0:
                # First segment can be 1-3 digits
                if len(seg) < 1 or len(seg) > 3:
                    raise ValueError(f"First segment must be 1-3 digits, got {len(seg)}")
            else:
                # Subsequent segments must be exactly 3 digits
                if len(seg) != 3:
                    raise ValueError(f"Comma-separated group must be 3 digits, got {len(seg)}")
        
        integer_part = no_comma
    else:
        # No commas, just digits
        if not integer_part.isdigit():
            raise ValueError(f"Invalid characters in integer part: {integer_part}")
        if len(integer_part) == 0:
            raise ValueError("No digits in integer part")
    
    # Combine integer and fractional parts
    total_cents_str = integer_part + fractional_part
    
    if not total_cents_str:
        raise ValueError("No digits found")
    
    total_cents = int(total_cents_str)
    
    if negative:
        total_cents = -total_cents
    
    return total_cents
