def round_nearest(x: float) -> int:
    """
    Round x to the nearest whole number and return it as an int.
    Python's built-in round() function implements 'round half to even'.
    For standard rounding (rounding 0.5 up), we can use the following logic.
    However, the examples 7.9 -> 8 and -3.2 -> -3 are standard.
    """
    return int(round(x))
