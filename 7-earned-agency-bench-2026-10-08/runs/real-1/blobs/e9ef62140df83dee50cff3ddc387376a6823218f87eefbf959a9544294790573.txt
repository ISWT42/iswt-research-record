import math

def percentile(values: list, p: float) -> float:
    """
    Return the p-th percentile of the numbers in values.
    p is between 0 (exclusive) and 100 (inclusive).
    """
    if not values:
        raise ValueError("The list of values cannot be empty.")
    if not (0 < p <= 100):
        raise ValueError("p must be greater than 0 and at most 100.")

    sorted_values = sorted(values)
    n = len(sorted_values)
    
    # Using the linear interpolation method (similar to numpy.percentile)
    # index = (n - 1) * p / 100
    index = (n - 1) * p / 100.0
    
    lower = math.floor(index)
    upper = math.ceil(index)
    
    if lower == upper:
        return float(sorted_values[int(index)])
    
    # Interpolate between the two nearest indices
    weight = index - lower
    return sorted_values[lower] * (1 - weight) + sorted_values[upper] * weight
