import re

_PATTERN = re.compile(r'^(\d{4})-(\d{2})-(\d{2})$')


def _parse(s: str):
    if not isinstance(s, str):
        raise ValueError("not a string")
    m = _PATTERN.match(s)
    if not m:
        raise ValueError("bad format")
    year, month, day = int(m.group(1)), int(m.group(2)), int(m.group(3))
    if not (1 <= month <= 12):
        raise ValueError("bad month")
    # days in month
    if month in (1, 3, 5, 7, 8, 10, 12):
        max_day = 31
    elif month in (4, 6, 9, 11):
        max_day = 30
    else:
        # February
        leap = (year % 4 == 0 and year % 100 != 0) or (year % 400 == 0)
        max_day = 29 if leap else 28
    if not (1 <= day <= max_day):
        raise ValueError("bad day")
    return year, month, day


def _to_ordinal(year: int, month: int, day: int) -> int:
    # days since 0001-01-01 (proleptic Gregorian), using civil_from_days style algorithm
    y = year
    m = month
    d = day
    y -= m <= 2
    era = (y if y >= 0 else y - 399) // 400
    yoe = y - era * 400
    doy = (153 * (m + (-3 if m > 2 else 9)) + 2) // 5 + d - 1
    doe = yoe * 365 + yoe // 4 - yoe // 100 + doy
    return era * 146097 + doe - 719468


def days_between(a: str, b: str) -> int:
    ya, ma, da = _parse(a)
    yb, mb, db = _parse(b)
    return _to_ordinal(yb, mb, db) - _to_ordinal(ya, ma, da)
